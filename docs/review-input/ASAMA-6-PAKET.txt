  602|     def cloud_upload_raw_content(self, filename: str, content: str, vehicle_vin: str | None = None) -> dict[str, Any]:
  603|         import os as _os
  604|         try:

  609|             if len(raw) > self._RAW_UPLOAD_MAX_BYTES:
  610|                 logger.warning("cloud_upload_raw_content rejected: oversize", extra={"bytes": len(raw)})
  611|                 return {"success": False, "error": "İçerik 1MB sınırını aşıyor."}

  617|                 "cloud_upload_raw_content accepted",
  618|                 extra={"upload_filename": clean_name, "bytes": len(raw)},
  619|             )
  620|             # Secure staging: O_CREAT|O_EXCL + 0o600 inside the app exports
  621|             # tree (no world-readable system-temp file, no delete=False race).

  627|             tmp_path: Path | None = None

  630|                 try:
  631|                     fd = _os.open(str(candidate), _os.O_CREAT | _os.O_EXCL | _os.O_WRONLY, 0o600)
  632|                 except FileExistsError:
  633|                     continue

  635|                     with _os.fdopen(fd, "wb") as fh:
  636|                         fh.write(raw)
  637|                 except Exception:
  638|                     try:
  639|                         candidate.unlink(missing_ok=True)
  640|                     except Exception:
  641|                         pass
  642|                     raise

  644|                 break

  646|                 return {"success": False, "error": "Geçici dosya oluşturulamadı."}
  647|             try:

  649|                 return {
  650|                     "success": True,
  651|                     "sessionId": result.session_id,
  652|                     "status": result.status,
  653|                 }
  654|             finally:
  655|                 try:
  656|                     tmp_path.unlink(missing_ok=True)
  657|                 except Exception:
  658|                     pass
  659|         except Exception as exc:
  660|             return {"success": False, "error": str(exc)}

 1182|     def _set_driver_listen_only(self, listen_only: bool) -> bool:
 1183|         """Flip the physical driver's listen-only mode (verified, best-effort).

 1419|     # Diagnostic Challenge & Action Execution Subsystem (Dual Confirmation)
 1420|     # ------------------------------------------------------------------
 1421|     def request_diagnostic_challenge(self, action: dict[str, Any]) -> dict[str, Any]:
 1422|         """Issue a short-lived (≤30s) single-use cryptographic confirmation token for a diagnostic action."""
 1423|         if not isinstance(action, dict):
 1424|             return {"success": False, "error": "Geçersiz aksiyon verisi (dictionary bekleniyor)."}
 1425| 
 1426|         action_type = str(action.get("action_type") or "").strip()
 1427|         if not action_type:
 1428|             return {"success": False, "error": "Aksiyon türü (action_type) belirtilmelidir."}
 1429| 
 1430|         action_id = str(action.get("id") or "")
 1431|         token = secrets.token_hex(16)
 1432|         now_ns = time.monotonic_ns()
 1433| 
 1434|         challenge = DiagnosticChallenge(
 1435|             token=token,
 1436|             action_type=action_type,
 1437|             action_id=action_id,
 1438|             created_at_monotonic_ns=now_ns,
 1439|             max_age_ns=30_000_000_000,
 1440|         )
 1441| 
 1442|         with self._challenges_lock:
 1443|             # Prune expired tokens
 1444|             expired = [k for k, ch in self._diagnostic_challenges.items() if (now_ns - ch.created_at_monotonic_ns) > ch.max_age_ns]
 1445|             for k in expired:
 1446|                 self._diagnostic_challenges.pop(k, None)
 1447|             self._diagnostic_challenges[token] = challenge
 1448| 
 1449|         return {
 1450|             "success": True,
 1451|             "token": token,
 1452|             "expires_in_s": 30.0,
 1453|             "action_type": action_type,
 1454|             "action_id": action_id,
 1455|         }

 1457|     def _verify_and_consume_diagnostic_token(self, token: str | None, action_type: str, action_id: str = "") -> tuple[bool, str]:
 1458|         """Verify that a single-use token is present, unexpired (≤30s), and matches action_type/action_id."""
 1459|         if not token or not isinstance(token, str) or not token.strip():
 1460|             return False, "Kullanıcı onayı gereklidir (Dual Confirmation challenge token eksik)."
 1461| 
 1462|         cleaned_token = token.strip()
 1463|         now_ns = time.monotonic_ns()
 1464| 
 1465|         with self._challenges_lock:
 1466|             challenge = self._diagnostic_challenges.pop(cleaned_token, None)
 1467|             if challenge is None:
 1468|                 return False, "Geçersiz veya daha önce kullanılmış onay token'ı (tek kullanımlık)."
 1469| 
 1470|             if (now_ns - challenge.created_at_monotonic_ns) > challenge.max_age_ns:
 1471|                 return False, "Onay token'ının süresi dolmuş (≤30s limit)."
 1472| 
 1473|             if challenge.action_type != action_type:
 1474|                 return False, f"Onay token'ı aksiyon türü ile uyuşmuyor ({challenge.action_type} != {action_type})."
 1475| 
 1476|             if action_id and challenge.action_id and challenge.action_id != action_id:
 1477|                 return False, f"Onay token'ı aksiyon kimliği ile uyuşmuyor ({challenge.action_id} != {action_id})."
 1478| 
 1479|         return True, "OK"

 1481|     def execute_diagnostic_action(
 1482|         self,
