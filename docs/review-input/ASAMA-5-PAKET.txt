   60| def _get_hmac_key(secret_provider: SecretProvider) -> bytes:
   61|     """Resolve the chunk-authentication HMAC key (E11 key-loss contract).
   62| 
   63|     Key loss semantics: if the vault key vanishes (reinstall/reset), a NEW key
   64|     is minted here. Chunks signed by the old key then fail HMAC verification
   65|     and are treated as unauthenticated — the legacy-file sweep at startup moves
   66|     them out of the active store instead of serving them. Recorded telemetry is
   67|     therefore never silently trusted across a key change: old chunks are kept
   68|     on disk under a quarantine name, new chunks authenticate normally.
   69|     """
   70|     try:
   71|         key = secret_provider.get_secret(HMAC_KEY_NAME)
   72|     except (KeyError, SecurityError):
   73|         key = os.urandom(32)
   74|         try:
   75|             secret_provider.store_secret(HMAC_KEY_NAME, key)
   76|         except Exception as store_err:
   77|             logger.warning("Failed to store rolling disk HMAC key", extra={"error": str(store_err)})
   78|         logger.info("Initialized rolling disk HMAC key")
   79| 
   80|     if len(key) != 32:
   81|         raise SecurityError("Rolling disk HMAC key has invalid length", code="SECURITY_ERROR")
   82|     return key

   91| def _check_frame(frame: CanFrame) -> None:
   92|     """Validate frame serializability WITHOUT packing (hot-path admission check).
   93| 
   94|     Perf: append() used to run _serialize_frame (a full struct.pack) purely to
   95|     validate, then flush() packed the same bytes again — 2x serialization per
   96|     frame on the RX path. This check enforces the exact same invariants as
   97|     _serialize_frame's raises, without building the 113-byte record.
   98| 
   99|     REVIEW 3-MEDIUM: enum membership (error_state/source/direction) is
  100|     verified at ADMISSION time — a frame mutated past __post_init__ to an
  101|     unmapped enum value previously slipped through admission and then raised
  102|     KeyError at flush, dropping the whole pending chunk. Rejecting the
  103|     single bad frame here keeps the chunk (and the black-box recording) alive.
  104|     """
  105|     if len(frame.channel_id.encode("utf-8")) > CHANNEL_ID_SIZE:
  106|         raise ValueError(f"channel_id exceeds {CHANNEL_ID_SIZE} UTF-8 bytes")
  107|     if len(frame.data) > DATA_SIZE:
  108|         raise ValueError(f"CAN payload exceeds {DATA_SIZE} bytes")
  109|     if not -(1 << 63) <= frame.sequence < (1 << 63):
  110|         raise ValueError("sequence must fit in a signed 64-bit integer")
  111|     if not 0 <= frame.timestamp_ns <= 0xFFFFFFFFFFFFFFFF:
  112|         raise ValueError("timestamp_ns must fit in an unsigned 64-bit integer")
  113|     if frame.hardware_timestamp_ns is not None and not 0 <= frame.hardware_timestamp_ns <= 0xFFFFFFFFFFFFFFFF:
  114|         raise ValueError("hardware_timestamp_ns must fit in an unsigned 64-bit integer")
  115|     if frame.host_timestamp_ns is not None and not 0 <= frame.host_timestamp_ns <= 0xFFFFFFFFFFFFFFFF:
  116|         raise ValueError("host_timestamp_ns must fit in an unsigned 64-bit integer")
  117|     if frame.error_state not in _ERROR_STATE_TO_CODE:
  118|         raise ValueError(f"unmapped error_state {frame.error_state!r}")
  119|     if frame.source not in _SOURCE_TO_CODE:
  120|         raise ValueError(f"unmapped source {frame.source!r}")

  123| def _serialize_frame(frame: CanFrame) -> bytes:
  124|     channel = frame.channel_id.encode("utf-8")
  125|     if len(channel) > CHANNEL_ID_SIZE:
  126|         raise ValueError(f"channel_id exceeds {CHANNEL_ID_SIZE} UTF-8 bytes")
  127|     if len(frame.data) > DATA_SIZE:
  128|         raise ValueError(f"CAN payload exceeds {DATA_SIZE} bytes")
  129|     if not -(1 << 63) <= frame.sequence < (1 << 63):
  130|         raise ValueError("sequence must fit in a signed 64-bit integer")
  131| 
  132|     flags = 0
  133|     if frame.is_extended:
  134|         flags |= FLAG_IS_EXTENDED
  135|     if frame.is_fd:
  136|         flags |= FLAG_IS_FD
  137|     if frame.brs:
  138|         flags |= FLAG_BRS
  139|     if frame.esi:
  140|         flags |= FLAG_ESI
  141|     if frame.direction == "tx":
  142|         flags |= FLAG_DIRECTION_TX
  143|     if frame.hardware_timestamp_ns is not None:
  144|         flags |= FLAG_HAS_HARDWARE_TIMESTAMP
  145|     if frame.host_timestamp_ns is not None:
  146|         flags |= FLAG_HAS_HOST_TIMESTAMP
  147| 
  148|     hardware_timestamp_ns = 0
  149|     if frame.hardware_timestamp_ns is not None:
  150|         hardware_timestamp_ns = _checked_u64(frame.hardware_timestamp_ns, "hardware_timestamp_ns")
  151|     host_timestamp_ns = 0
  152|     if frame.host_timestamp_ns is not None:
  153|         host_timestamp_ns = _checked_u64(frame.host_timestamp_ns, "host_timestamp_ns")
  154| 
  155|     return struct.pack(
  156|         FRAME_FMT,
  157|         frame.arbitration_id,
  158|         _checked_u64(frame.timestamp_ns, "timestamp_ns"),
  159|         hardware_timestamp_ns,
  160|         host_timestamp_ns,
  161|         frame.sequence,
  162|         flags,
  163|         frame.dlc,
  164|         _ERROR_STATE_TO_CODE[frame.error_state],
  165|         _SOURCE_TO_CODE[frame.source],
  166|         len(channel),
  167|         0,
  168|         channel.ljust(CHANNEL_ID_SIZE, b"\x00"),
  169|         len(frame.data),
  170|         frame.data.ljust(DATA_SIZE, b"\xcc"),
  171|     )

  174| def _serialize_chunk(frames: list[CanFrame], key: bytes) -> bytes:
  175|     if len(key) != 32:
  176|         raise SecurityError("Rolling disk HMAC key has invalid length", code="SECURITY_ERROR")
  177|     if len(frames) > 0xFFFFFFFF:
  178|         raise ValueError("Chunk frame count exceeds format limit")
  179| 
  180|     body = b"".join(_serialize_frame(frame) for frame in frames)
  181|     prefix = struct.pack(
  182|         HEADER_PREFIX_FMT,
  183|         CHUNK_MAGIC,
  184|         CHUNK_VERSION,
  185|         0,
  186|         len(frames),
  187|         FRAME_SIZE,
  188|     )
  189|     mac = hmac.new(key, prefix + body, hashlib.sha256).digest()
  190|     return prefix + mac + body

  247| def _deserialize_chunk(raw: bytes, key: bytes) -> list[CanFrame]:
  248|     if len(key) != 32:
  249|         raise SecurityError("Rolling disk HMAC key has invalid length", code="SECURITY_ERROR")
  250|     if len(raw) < HEADER_SIZE:
  251|         raise SecurityError("Rolling disk chunk is too short", code="SECURITY_ERROR")
  252| 
  253|     magic, version, header_flags, count, record_size, stored_mac = struct.unpack(
  254|         HEADER_FMT, raw[:HEADER_SIZE]
  255|     )
  256|     if magic != CHUNK_MAGIC:
  257|         raise SecurityError("Invalid rolling disk chunk magic", code="SECURITY_ERROR")
  258|     if version != CHUNK_VERSION:
  259|         raise SecurityError(f"Unsupported rolling disk chunk version {version}", code="SECURITY_ERROR")
  260|     if header_flags != 0:
  261|         raise SecurityError("Reserved chunk header flags are set", code="SECURITY_ERROR")
  262|     if record_size != FRAME_SIZE:
  263|         raise SecurityError("Unexpected rolling disk frame size", code="SECURITY_ERROR")
  264| 
  265|     expected_size = HEADER_SIZE + count * FRAME_SIZE
  266|     if len(raw) != expected_size:
  267|         raise SecurityError("Rolling disk chunk length does not match its header", code="SECURITY_ERROR")
  268| 
  269|     prefix = raw[: struct.calcsize(HEADER_PREFIX_FMT)]
  270|     body = raw[HEADER_SIZE:]
  271|     calculated_mac = hmac.new(key, prefix + body, hashlib.sha256).digest()
  272|     if not hmac.compare_digest(calculated_mac, stored_mac):
  273|         raise SecurityError("Rolling disk chunk HMAC mismatch", code="SECURITY_ERROR")
  274| 
  275|     return [
  276|         _deserialize_frame(body[offset : offset + FRAME_SIZE])
  277|         for offset in range(0, len(body), FRAME_SIZE)
  278|     ]

  293|     def __init__(
  294|         self,
  295|         storage_dir: str | Path,
  296|         max_retention_sec: int = MAX_RETENTION_SEC,
  297|         max_disk_bytes: int = MAX_DISK_BYTES,
  298|         chunk_frame_threshold: int = CHUNK_THRESHOLD_FRAMES,
  299|         secret_provider: SecretProvider | None = None,
  300|     ) -> None:
  301|         if (
  302|             not isinstance(chunk_frame_threshold, int)
  303|             or isinstance(chunk_frame_threshold, bool)
  304|             or not (1 <= chunk_frame_threshold <= self.MAX_CHUNK_FRAMES)
  305|         ):
  306|             raise ValueError(
  307|                 f"chunk_frame_threshold must be in range 1..{self.MAX_CHUNK_FRAMES}, "
  308|                 f"got {chunk_frame_threshold!r}"
  309|             )
  310| 
  311|         raw_dir = Path(storage_dir)
  312|         if raw_dir.is_symlink():
  313|             raise ValueError(f"storage_dir must not be a symlink, got {storage_dir!r}")
  314|         raw_dir.mkdir(parents=True, exist_ok=True)
  315|         if raw_dir.is_symlink():
  316|             raise ValueError(f"storage_dir must not be a symlink, got {storage_dir!r}")
  317|         self.storage_dir = raw_dir.resolve()
  318|         if os.name != "nt":
  319|             try:
  320|                 os.chmod(self.storage_dir, 0o700)
  321|             except OSError:
  322|                 pass
  323|         self.max_retention_sec = max_retention_sec
  324|         self.max_disk_bytes = max_disk_bytes
  325|         self.chunk_frame_threshold = chunk_frame_threshold
  326|         self._secret_provider = secret_provider or get_default_secret_provider()
  327| 
  328|         self._lock = threading.RLock()
  329|         self._tls = threading.local()
  330|         self._closed = False
  331|         self._current_chunk_frames: list[CanFrame] = []
  332|         # E10: count of frames rejected at append-time for malformed content
  333|         self._rejected_frames: int = 0
  334|         self._cctx = zstd.ZstdCompressor(level=3)
  335|         self._dctx = zstd.ZstdDecompressor()
  336|         max_read_frames = max(chunk_frame_threshold, self.CHUNK_THRESHOLD_FRAMES)
  337|         self._max_chunk_bytes = min(HEADER_SIZE + max_read_frames * FRAME_SIZE, self.MAX_CHUNK_BYTES)
  338|         self._migrate_legacy_chunks()
  339|         self._chunk_index = self._next_chunk_index()
  340|         # F-34: bounded async write queue + worker thread. Serialize + HMAC stay
  341|         # on the caller thread (single-writer ordering); only compression and

  343|         # (documented drop-free fallback, preferable to losing telemetry).
  344|         self._flush_queue: queue.Queue[tuple[Path, bytes] | None] = queue.Queue(maxsize=16)
  345|         self._flush_worker = threading.Thread(
  346|             target=self._flush_worker_loop, name="rolling_disk_flush", daemon=True

  348|         self._flush_worker.start()

  386|     def append(self, frame: CanFrame) -> None:
  387|         """Add a frame and flush when the configured chunk threshold is reached.
  388| 
  389|         E10: a malformed frame (oversized channel/payload, out-of-range field)
  390|         would raise ValueError from serialization at flush time and kill the
  391|         ingestion loop. Reject it here, before it enters the chunk, and keep
  392|         the blackbox recording — one bad frame must not stop the recorder.
  393| 
  394|         Perf: admission runs _check_frame (pure field-range checks, no
  395|         struct.pack); the actual bytes are built once, at flush() time.
  396|         """
  397|         try:
  398|             _check_frame(frame)
  399|         except (ValueError, KeyError) as exc:
  400|             # M-21 (P2-19): KeyError joins the guard — an unmapped
  401|             # error_state/source value must reject the frame, not kill the
  402|             # ingestion loop at flush time.
  403|             with self._lock:
  404|                 self._rejected_frames += 1
  405|             logger.warning(
  406|                 "Rolling disk rejected malformed frame",
  407|                 extra={"error": str(exc), "total_rejected": self._rejected_frames},
  408|             )
  409|             return
  410| 
  411|         with self._lock:
  412|             # REVIEW (close-vs-append race): after close() the flush worker
  413|             # is gone — appended frames would sit in _current_chunk_frames
  414|             # forever and silently never reach disk (black-box durability
  415|             # violation), and drain calls would block 30 s against a dead
  416|             # worker. Reject late frames loudly, once per chunk, and count.
  417|             if self._closed:
  418|                 self._rejected_frames += 1
  419|                 logger.error(
  420|                     "Rolling disk closed — frame rejected (data loss window)",
  421|                     extra={"arbitration_id": getattr(frame, "arbitration_id", None), "total_rejected": self._rejected_frames},
  422|                 )
  423|                 return
  424|             self._current_chunk_frames.append(frame)
  425|             should_flush = len(self._current_chunk_frames) >= self.chunk_frame_threshold
  426|         if should_flush:
  427|             self.flush(drain=(self.chunk_frame_threshold <= 10))

  429|     def flush(self, drain: bool = True) -> Path | None:
  430|         """Authenticate and enqueue the pending chunk for async disk write (F-34).
  431| 
  432|         Serialization + HMAC run on the caller thread (single-writer
  433|         ordering); compression and disk IO are offloaded to the worker.
  434|         Queue full => synchronous fallback write (documented in module doc).
  435|         """
  436|         with self._lock:
  437|             if not self._current_chunk_frames:
  438|                 return None
  439|             frames = self._current_chunk_frames
  440|             self._current_chunk_frames = []
  441|             chunk_idx = self._chunk_index
  442|             self._chunk_index += 1
  443| 
  444|         key = _get_hmac_key(self._secret_provider)
  445|         try:
  446|             raw_bytes = self._serialize_chunk(frames, key)
  447|         except (ValueError, KeyError, struct.error) as exc:
  448|             # E10 defense-in-depth: append() already validates frames, but a
  449|             # frame mutated after admission must not kill the ingestion loop —
  450|             # drop the whole pending chunk (it cannot be partially trusted)
  451|             # and keep recording the next one. M-21 (P2-19): KeyError and
  452|             # struct.error join the guard for the same reason.
  453|             with self._lock:
  454|                 self._rejected_frames += len(frames)
  455|             logger.error(
  456|                 "Rolling disk dropped unserializable pending chunk",
  457|                 extra={"error": str(exc), "frames": len(frames)},
  458|             )
  459|             return None
  460|         timestamp_ns = frames[0].timestamp_ns
  461|         chunk_file = self.storage_dir / (
  462|             f"chunk_{chunk_idx:08d}_{timestamp_ns:020d}.bin.zst"
  463|         )
  464| 
  465|         try:
  466|             self._flush_queue.put_nowait((chunk_file, raw_bytes))
  467|         except queue.Full:
  468|             # Bounded queue full: write synchronously rather than drop frames
  469|             self._write_chunk(chunk_file, raw_bytes)
  470|         else:
  471|             if drain:
  472|                 self._drain_flush_queue(timeout_s=30.0)
  473| 
  474|         logger.debug(
  475|             "Enqueued rolling disk chunk",
  476|             extra={
  477|                 "file": chunk_file.name,
  478|                 "frames": len(frames),
  479|                 "uncompressed_kb": len(raw_bytes) // 1024,
  480|             },
  481|         )
  482|         return chunk_file

  484|     def _write_chunk(self, chunk_file: Path, raw_bytes: bytes) -> bytes:
  485|         """Compress and atomically persist one serialized chunk; returns bytes written.
  486| 
  487|         E-C-003: durability sequence write -> flush -> fsync -> replace, so
  488|         a power cut after the rename can never leave a truncated chunk
  489|         shadowing the kara kutu (black-box) recording.
  490|         """
  491|         cctx = getattr(self._tls, "cctx", None)
  492|         if cctx is None:
  493|             cctx = zstd.ZstdCompressor(level=3)
  494|             self._tls.cctx = cctx
  495|         compressed_bytes = cctx.compress(raw_bytes)
  496|         temporary_file = chunk_file.with_name(
  497|             f"{chunk_file.name}.{os.getpid()}_{threading.get_ident()}_{time.monotonic_ns()}.tmp"
  498|         )
  499|         flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL
  500|         # O_EXCL guarantees no TOCTOU overwrite of a planted symlink/file.

   23| @dataclass(slots=True)
   24| class Subscription:
   25|     """Active subscriber registration handle."""
   26| 
   27|     sub_id: int
   28|     callback: Callable[[CanFrame], None] | None = None
   29|     frame_queue: queue.Queue[CanFrame] | None = None
   30|     filter_ids: set[int] | None = None  # None = accept all arbitration IDs
   31|     channel_id: str | None = None  # None = accept all channels
   32|     is_demoted: bool = False
   33| 
   34| 
   35| class FrameRouter:
   36|     """Centralized thread-safe message router and dispatcher for CAN bus streams."""
   37| 
   38|     MAX_QUEUE_SIZE: ClassVar[int] = 10_000
   39| 
   40|     # M-7 (3FABLE): a synchronous callback running longer than this budget
   41|     # is demoted — a WARN is emitted and the subscriber is moved to
   42|     # queue-only dispatch so one slow consumer (UI, AI copilot) can never
   43|     # HOL-block the RX thread and trip protocol timers (N_Cr, T1).
   44|     CALLBACK_BUDGET_MS: ClassVar[float] = 20.0
   45|     # E-2: rate-limited drop logging — one summary per subscriber per second.
   46|     _DROP_LOG_INTERVAL_S: ClassVar[float] = 1.0
   47|     # REVIEW2 #2 / REVIEW3 #13: a callback-ONLY subscriber (no queue to be
   48|     # demoted onto) that keeps blowing the budget gets its callback REMOVED
   49|     # after this many consecutive violations — "deaf" beats "holds the whole
   50|     # RX thread hostage on every frame" (protocol timers miss otherwise).
   51|     CALLBACK_TRIP_AFTER: ClassVar[int] = 10
   52| 
   53|     def __init__(self) -> None:
   54|         self._lock = threading.Lock()
   55|         self._subscriptions: dict[int, Subscription] = {}
   56|         # Perf (C-8): immutable routing snapshot — route_frame() fans frames
   57|         # out over this tuple WITHOUT taking the lock or copying the dict.
   58|         # Mutators (subscribe/unsubscribe/restore/clear) rebuild the snapshot
   59|         # under the lock (copy-on-write). Subscriptions change at setup/teardown
   60|         # rates, frames at bus rates — the trade favors the hot path.
   61|         self._route_snapshot: tuple[Subscription, ...] = ()
   62|         self._next_sub_id: int = 1
   63|         # L-16 (P3-11): dedicated counter lock. The counters were mutated
   64|         # without any lock; multiple ingest threads racing `+=` lost
   65|         # increments (metrics only — never frame delivery, which iterates
   66|         # the immutable copy-on-write snapshot and needs no lock).
   67|         self._stats_lock = threading.Lock()
   68|         self._total_routed: int = 0
   69|         self._total_dropped: int = 0
   70|         # E-2: (sub_id, monotonic-second) of the last drop log per subscriber
   71|         self._last_drop_log: dict[int, float] = {}
   72|         self._drop_counts_since_log: dict[int, int] = {}
   73|         # REVIEW2 #2: consecutive CALLBACK_BUDGET_MS violations per subscriber
   74|         # (callback-only subscribers get tripped; queue subscribers get demoted).
   75|         self._callback_trip_counts: dict[int, int] = {}
   76| 
   77|     def _rebuild_snapshot_locked(self) -> None:
   78|         """Refresh the lock-free routing snapshot. Caller must hold the lock."""
   79|         self._route_snapshot = tuple(self._subscriptions.values())
   80| 
   81|     def subscribe(
   82|         self,
   83|         callback: Callable[[CanFrame], None] | None = None,
   84|         filter_ids: set[int] | None = None,
   85|         channel_id: str | None = None,
   86|         use_queue: bool = False,
   87|         queue_maxsize: int = MAX_QUEUE_SIZE,
   88|     ) -> tuple[int, queue.Queue[CanFrame] | None]:
   89|         """Register a new consumer subscription.
   90| 
   91|         Returns (subscription_id, queue_instance_or_None).
   92|         """
   93|         if (
   94|             not isinstance(queue_maxsize, int)
   95|             or isinstance(queue_maxsize, bool)
   96|             or not (1 <= queue_maxsize <= self.MAX_QUEUE_SIZE)
   97|         ):
   98|             raise ValueError(
   99|                 f"queue_maxsize must be in range 1..{self.MAX_QUEUE_SIZE}, got {queue_maxsize!r}"
  100|             )
  101|         fq: queue.Queue[CanFrame] | None = None
  102|         if use_queue:
  103|             fq = queue.Queue(maxsize=queue_maxsize)
  104| 
  105|         with self._lock:
  106|             sub_id = self._next_sub_id
  107|             self._next_sub_id += 1
  108|             sub = Subscription(
  109|                 sub_id=sub_id,
  110|                 callback=callback,
  111|                 frame_queue=fq,
  112|                 filter_ids=set(filter_ids) if filter_ids is not None else None,
  113|                 channel_id=channel_id,
  114|             )
  115|             self._subscriptions[sub_id] = sub
  116|             self._rebuild_snapshot_locked()
  117| 
  118|         return sub_id, fq
  119| 
  120|     def unsubscribe(self, sub_id: int) -> bool:
  121|         """Remove an existing subscription by ID."""
  122|         with self._lock:
  123|             removed = self._subscriptions.pop(sub_id, None) is not None
  124|             if removed:
  125|                 self._rebuild_snapshot_locked()
  126|                 # REVIEW2 #2: drop per-subscriber bookkeeping with the sub.
  127|                 self._callback_trip_counts.pop(sub_id, None)
  128|                 self._drop_counts_since_log.pop(sub_id, None)
  129|                 self._last_drop_log.pop(sub_id, None)
  130|             return removed
  131| 
  132|     def route_frame(self, frame: CanFrame) -> int:
  133|         """Dispatch an ingested frame to all matching subscribers.
  134| 
  135|         Returns the number of subscribers that accepted the frame.
  136| 
  137|         Perf (C-8): iterates the immutable copy-on-write snapshot — no lock
  138|         acquisition, no per-frame list(values()) copy. In-flight frames may
  139|         see a subscription removed microseconds earlier; that is the same
  140|         linearization the old lock+copy provided, at a fraction of the cost.
  141|         """
  142|         subscribers = self._route_snapshot
  143|         with self._stats_lock:  # L-16: metrics-scope lock, hot path safe
  144|             self._total_routed += 1
  145| 
  146|         matched_count = 0
  147|         for sub in subscribers:
  148|             # Check channel filter
  149|             if sub.channel_id is not None and frame.channel_id != sub.channel_id:
  150|                 continue
  151| 
  152|             # Check arbitration ID filter
  153|             if sub.filter_ids is not None and frame.arbitration_id not in sub.filter_ids:
  154|                 continue
  155| 
  156|             matched_count += 1
  157| 
  158|             # M-7: dispatch to callback with a time budget; overspending
  159|             # subscribers are demoted to queue-only so they stop stalling
  160|             # the RX thread.
  161|             if sub.callback is not None:
  162|                 t0 = time.monotonic()
  163|                 try:
  164|                     sub.callback(frame)
  165|                 except Exception as exc:  # noqa: BLE001
  166|                     logger.error(
  167|                         "Error in FrameRouter subscriber callback",
  168|                         extra={"sub_id": sub.sub_id, "error": str(exc)},
  169|                     )
  170|                 elapsed_ms = (time.monotonic() - t0) * 1000.0
  171|                 if elapsed_ms > self.CALLBACK_BUDGET_MS:
  172|                     logger.warning(
  173|                         "Slow FrameRouter callback exceeded time budget",
  174|                         extra={
  175|                             "sub_id": sub.sub_id,
  176|                             "elapsed_ms": round(elapsed_ms, 2),
  177|                             "budget_ms": self.CALLBACK_BUDGET_MS,
  178|                         },
  179|                     )
  180|                     # Only demote to queue-only if the subscriber actually has a queue to read from.
  181|                     # Setting callback=None on a callback-only subscriber leaves it completely deaf.
  182|                     if sub.frame_queue is not None:
  183|                         with self._lock:
  184|                             if sub.sub_id in self._subscriptions:
  185|                                 self._subscriptions[sub.sub_id].callback = None
  186|                                 self._subscriptions[sub.sub_id].is_demoted = True
  187|                     else:
  188|                         # REVIEW2 #2 / REVIEW3 #13: a queue-less subscriber
  189|                         # cannot be demoted, but it also must not keep
  190|                         # stalling every routed frame forever. After
  191|                         # CALLBACK_TRIP_AFTER consecutive over-budget calls
  192|                         # the callback is removed (subscriber goes deaf,
  193|                         # loudly logged) — restore_callback() can re-arm it.
  194|                         with self._lock:
  195|                             trips = self._callback_trip_counts.get(sub.sub_id, 0) + 1
  196|                             self._callback_trip_counts[sub.sub_id] = trips
  197|                             if trips >= self.CALLBACK_TRIP_AFTER and sub.sub_id in self._subscriptions:
  198|                                 self._subscriptions[sub.sub_id].callback = None
  199|                                 self._subscriptions[sub.sub_id].is_demoted = True
  200|                                 logger.error(
  201|                                     "FrameRouter callback-only subscriber tripped after repeated budget violations — "
  202|                                     "callback removed (restore via restore_callback)",
  203|                                     extra={"sub_id": sub.sub_id, "consecutive_violations": trips},
  204|                                 )
  205|                 else:
  206|                     # In-budget call resets the consecutive-violation counter.
  207|                     if self._callback_trip_counts:
  208|                         with self._lock:
  209|                             self._callback_trip_counts.pop(sub.sub_id, None)
  210| 
  211|             # Dispatch to queue (non-blocking with drop on full)
  212|             if sub.frame_queue is not None:
  213|                 try:
  214|                     sub.frame_queue.put_nowait(frame)
  215|                 except queue.Full:
  216|                     with self._lock:
  217|                         self._total_dropped += 1
  218|                         self._drop_counts_since_log[sub.sub_id] = (
  219|                             self._drop_counts_since_log.get(sub.sub_id, 0) + 1
  220|                         )
  221|                         now = time.monotonic()
  222|                         should_log = (now - self._last_drop_log.get(sub.sub_id, 0.0)) >= self._DROP_LOG_INTERVAL_S
  223|                         if should_log:
  224|                             burst = self._drop_counts_since_log.pop(sub.sub_id, 0)
  225|                             self._last_drop_log[sub.sub_id] = now
  226|                     # E-2: log at most one summary per subscriber per second —
  227|                     # per-frame WARNINGs at bus rate flooded the log disk I/O
  228|                     # and slowed the RX thread further (positive feedback).
  229|                     if should_log:
  230|                         logger.warning(
  231|                             "FrameRouter subscriber queue full; frames dropped",
  232|                             extra={
  233|                                 "sub_id": sub.sub_id,
  234|                                 "dropped_in_window": burst,
  235|                                 "arbitration_id_last": hex(frame.arbitration_id),
  236|                             },
  237|                         )
  238| 
  239|         return matched_count
  240| 
  241|     def restore_callback(
  242|         self, sub_id: int, callback: Callable[[CanFrame], None]
  243|     ) -> bool:
  244|         """Restore or reassign callback on an existing (or demoted) subscription (MED-4)."""
  245|         with self._lock:
  246|             sub = self._subscriptions.get(sub_id)
  247|             if sub is None:
  248|                 return False
  249|             sub.callback = callback
  250|             sub.is_demoted = False
  251|             # REVIEW2 #2: a fresh operator-armed callback starts with a
  252|             # clean trip count (it may genuinely be faster now).
  253|             self._callback_trip_counts.pop(sub_id, None)
  254|             return True
  255| 
  256|     def clear(self) -> None:
  257|         """Remove all active subscriptions."""
  258|         with self._lock:
  259|             self._subscriptions.clear()
  260|             self._rebuild_snapshot_locked()
  261| 
  262|     @property
  263|     def subscription_count(self) -> int:
  264|         with self._lock:
  265|             return len(self._subscriptions)
  266| 
  267|     @property
  268|     def stats(self) -> dict[str, int]:
  269|         with self._lock:
  270|             return {
  271|                 "active_subscriptions": len(self._subscriptions),
  272|                 "total_routed": self._total_routed,
  273|                 "total_dropped": self._total_dropped,
  274|             }
