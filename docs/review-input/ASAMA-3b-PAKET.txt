 2058|     @staticmethod
 2059|     def _levenshtein_distance(s1: str, s2: str) -> int:
 2060|         if abs(len(s1) - len(s2)) > 1:
 2061|             return 2
 2062|         if s1 == s2:
 2063|             return 0
 2064|         d: dict[tuple[int, int], int] = {}
 2065|         len1, len2 = len(s1), len(s2)
 2066|         for i in range(-1, len1 + 1):
 2067|             d[(i, -1)] = i + 1
 2068|         for j in range(-1, len2 + 1):
 2069|             d[(-1, j)] = j + 1
 2070|         for i in range(len1):
 2071|             for j in range(len2):
 2072|                 cost = 0 if s1[i] == s2[j] else 1
 2073|                 d[(i, j)] = min(
 2074|                     d[(i - 1, j)] + 1,
 2075|                     d[(i, j - 1)] + 1,
 2076|                     d[(i - 1, j - 1)] + cost
 2077|                 )
 2078|                 if i > 0 and j > 0 and s1[i] == s2[j - 1] and s1[i - 1] == s2[j]:
 2079|                     d[(i, j)] = min(d[(i, j)], d[(i - 2, j - 2)] + 1)
 2080|         return d[(len1 - 1, len2 - 1)]
