   53| class TxBudget:
   54|     """Monotonic token bucket: `capacity` burst tokens refilled at `refill_per_sec`."""
   55| 
   56|     __slots__ = ("capacity", "refill_per_sec", "_tokens", "_last_refill_ns", "_lock")
   57| 
   58|     def __init__(self, capacity: int, refill_per_sec: float) -> None:
   59|         self.capacity = capacity
   60|         self.refill_per_sec = refill_per_sec
   61|         self._tokens = float(capacity)
   62|         self._last_refill_ns = time.monotonic_ns()
   63|         self._lock = threading.Lock()
   64| 
   65|     def try_consume(self, n: int = 1) -> bool:
   66|         """Consume `n` tokens if available; returns False when the bucket is dry."""
   67|         with self._lock:
   68|             now_ns = time.monotonic_ns()
   69|             elapsed_s = (now_ns - self._last_refill_ns) / 1e9
   70|             if elapsed_s > 0:
   71|                 self._tokens = min(float(self.capacity), self._tokens + elapsed_s * self.refill_per_sec)
   72|                 self._last_refill_ns = now_ns
   73|             if self._tokens >= n:
   74|                 self._tokens -= n
   75|                 return True
   76|             return False
   77| 
   78|     def refund(self, n: int = 1) -> None:
   79|         """Refund `n` previously consumed tokens back to the bucket (capped at capacity)."""
   80|         with self._lock:
   81|             self._tokens = min(float(self.capacity), self._tokens + n)

  229|     def rebind_bus(self, new_bus: AbstractBus) -> None:
  230|         """Controlled HAL bus swap (reconnect path).
  231| 
  232|         The only sanctioned way to change the bus after construction. The swap
  233|         itself is explicit and auditable; arbitrary attribute assignment stays
  234|         blocked. Clears rate-limit state so the new channel starts clean.
  235| 
  236|         REVIEW (validate-then-dispatch TOCTOU, B-25): the swap runs under
  237|         the E-Stop TX send lock and bumps the fence generation. A frame
  238|         validated against the OLD bus (channel/whitelist/rate state) can no
  239|         longer dispatch onto the NEW bus mid-reconnect — the PHASE 3 fence
  240|         re-check rejects it fail-closed instead of writing to the wrong
  241|         physical network.
  242|         """
  243|         # Serialize against in-flight dispatches (fence re-check + send run
  244|         # under this lock) so the swap is atomic with respect to senders.
  245|         with self.estop.tx_send_lock:
  246|             with self._lock:
  247|                 old = self._bus
  248|                 self._bus = new_bus
  249|                 self._tx_timestamps.clear()
  250|                 self._rate_overload_streak = 0
  251|                 self._whitelist_miss_streak = 0
  252|             # Invalidate every in-flight validated frame: dispatch compares
  253|             # its fence snapshot under tx_send_lock and rejects on mismatch.
  254|             # estop.tx_fence getter takes the estop lock; bump via trigger-
  255|             # free generation advance (read-modify-write under estop lock).
  256|             self.estop.advance_tx_fence("gateway bus rebind")
  257|         logger.warning(
  258|             "TX Gateway bus rebound",
  259|             extra={
  260|                 "old_channel": getattr(old, "channel_id", None),
  261|                 "new_channel": getattr(new_bus, "channel_id", None),
  262|             },
  263|         )

  265|     def issue_confirmation_token(self, arbitration_id: int, ttl_s: float = 30.0) -> bytes:
  266|         """Mint a single-use HMAC confirmation token for a critical arbitration ID.
  267| 
  268|         Operator/UI authorization path: the token binds (arbitration_id, expiry,
  269|         nonce) under the gateway confirmation secret. Requires a configured secret.
  270|         """
  271|         if self._confirmation_secret is None:
  272|             raise SafetyError(
  273|                 "Confirmation tokens require a gateway confirmation_secret",
  274|                 code="CONFIRMATION_NOT_CONFIGURED",
  275|             )
  276|         import secrets as _secrets
  277| 
  278|         expiry_ns = time.monotonic_ns() + int(max(1.0, ttl_s) * 1_000_000_000)
  279|         nonce = _secrets.token_bytes(16)
  280|         payload = (
  281|             int(arbitration_id).to_bytes(4, "big") + expiry_ns.to_bytes(8, "big", signed=False) + nonce
  282|         )
  283|         mac = hmac.new(self._confirmation_secret, payload, hashlib.sha256).digest()
  284|         return payload + mac

  286|     def _verify_confirmation_token(self, token: bytes | str, frame: CanFrame) -> None:
  287|         """Verify a presented ConfirmationToken (fail-closed, single-use, TTL-bound)."""
  288|         assert self._confirmation_secret is not None
  289|         raw: bytes
  290|         if isinstance(token, str):
  291|             try:
  292|                 raw = bytes.fromhex(token.strip())
  293|             except ValueError as exc:
  294|                 raise DualConfirmationRequiredError(
  295|                     "Critical command rejected: malformed confirmation token",
  296|                 ) from exc
  297|         elif isinstance(token, (bytes, bytearray)):
  298|             raw = bytes(token)
  299|         else:
  300|             raise DualConfirmationRequiredError(
  301|                 "Critical command rejected: malformed confirmation token",
  302|             )
  303|         if len(raw) != 4 + 8 + 16 + 32:
  304|             raise DualConfirmationRequiredError(
  305|                 "Critical command rejected: malformed confirmation token",
  306|             )
  307|         payload, mac = raw[:-32], raw[-32:]
  308|         expected = hmac.new(self._confirmation_secret, payload, hashlib.sha256).digest()
  309|         if not hmac.compare_digest(mac, expected):
  310|             raise DualConfirmationRequiredError(
  311|                 "Critical command rejected: invalid confirmation token",
  312|             )
  313|         arb = int.from_bytes(payload[0:4], "big")
  314|         expiry_ns = int.from_bytes(payload[4:12], "big", signed=False)
  315|         if arb != frame.arbitration_id:
  316|             raise DualConfirmationRequiredError(
  317|                 "Critical command rejected: confirmation token arbitration mismatch",
  318|             )
  319|         if time.monotonic_ns() > expiry_ns:
  320|             raise DualConfirmationRequiredError(
  321|                 "Critical command rejected: confirmation token expired",
  322|             )
  323|         if payload in self._consumed_confirmations:
  324|             raise DualConfirmationRequiredError(
  325|                 "Critical command rejected: confirmation token already consumed",
  326|             )
  327|         self._consumed_confirmations.add(payload)
  328|         if len(self._consumed_confirmations) > 1024:
  329|             # Bounded replay set: TTL expiry bounds live tokens; evict oldest.
  330|             self._consumed_confirmations.pop()

  376|     def update_vehicle_speed(self, speed_kmh: float, *, source: str = "physical") -> None:
  377|         """Update live vehicle speed for dynamic interlock enforcement.
  378| 
  379|         Always timestamps with time.monotonic_ns() on reception to prevent
  380|         clock domain skew. NaN, negative or non-finite values are treated as
  381|         corrupted telemetry and invalidate freshness to 0 (fail-closed).
  382| 
  383|         P0-2 (REVIEW C-2): `source` binds provenance to the interlock. Only
  384|         `"physical"` (hardware-derived telemetry) may satisfy the speed
  385|         interlock; any other value is recorded for telemetry purposes but
  386|         NEVER refreshes interlock freshness — a simulated stationary vehicle
  387|         must not authorize critical commands while a real vehicle is
  388|         connected and moving.
  389| 
  390|         Backward compat: signature kept, but non-"physical" sources are
  391|         strictly non-authoritative. Prefer update_physical_speed() /
  392|         record_synthetic_speed() going forward.
  393|         """
  394|         with self._lock:
  395|             if not math.isfinite(speed_kmh) or speed_kmh < 0.0:
  396|                 self._current_vehicle_speed_kmh = float("nan")
  397|                 self._last_speed_update_ns = 0
  398|                 return
  399|             if source != "physical":
  400|                 # Tightened: any non-"physical" source (incl. "synthetic")
  401|                 # updates the DISPLAY value but never touches freshness. The
  402|                 # interlock then keeps evaluating against the last physical
  403|                 # sample (fail-closed when none ever arrived). This preserves
  404|                 # the existing P0-2 test contract: synthetic 25 km/h reads as
  405|                 # not-fresh-and-safe while physical timestamps stay pinned.
  406|                 self._current_vehicle_speed_kmh = float(speed_kmh)
  407|                 return
  408|             self._current_vehicle_speed_kmh = float(speed_kmh)
  409|             self._last_speed_update_ns = time.monotonic_ns()

  425|     def is_speed_fresh_and_safe(self, max_age_ns: int | None = None) -> bool:
  426|         """Public inquiry API for preflight checks: True when speed is fresh and below threshold."""
  427|         timeout = max_age_ns if max_age_ns is not None else self.SPEED_VALIDITY_TIMEOUT_NS
  428|         with self._lock:
  429|             if self._last_speed_update_ns == 0:
  430|                 return False
  431|             now_ns = time.monotonic_ns()
  432|             if (now_ns - self._last_speed_update_ns) > timeout:
  433|                 return False
  434|             if math.isnan(self._current_vehicle_speed_kmh):
  435|                 return False
  436|             return self._current_vehicle_speed_kmh <= self.SPEED_NOISE_THRESHOLD_KMH

  438|     def speed_interlock_state(self, max_age_ns: int | None = None) -> tuple[str, float]:
  439|         """REVIEW 3: tri-state interlock verdict with the live speed value.
  440| 
  441|         Single authoritative speed source for UI/supervisor preflight gates
  442|         (kills the desktop's parallel `_current_speed_kmh` truth):
  443|           ("stale", speed)  — no physical feed yet, timed out, or NaN
  444|           ("moving", speed) — fresh physical speed above the noise threshold
  445|           ("ok", speed)     — fresh, finite, at/below threshold
  446|         """
  447|         timeout = max_age_ns if max_age_ns is not None else self.SPEED_VALIDITY_TIMEOUT_NS
  448|         with self._lock:
  449|             speed = self._current_vehicle_speed_kmh
  450|             if self._last_speed_update_ns == 0:
  451|                 return "stale", speed
  452|             if (time.monotonic_ns() - self._last_speed_update_ns) > timeout:
  453|                 return "stale", speed
  454|             if math.isnan(speed):
  455|                 return "stale", speed
  456|             if speed > self.SPEED_NOISE_THRESHOLD_KMH:
  457|                 return "moving", speed
  458|             return "ok", speed

  340|     def create_reset_token(self) -> EmergencyStopToken | None:
  341|         """Generate a valid, signed EmergencyStopToken for the currently active challenge.
  342| 
  343|         P1-1 (self-signing separation): token MINTING is an authorization

  346|         component configured with `allow_self_reset=True`. Any code holding

  351|             if not self._allow_self_reset:
  352|                 raise SafetyError(

  355|                     code="ESTOP_MINT_DENIED",

  358|                 return None

  360|             now_ns = time.monotonic_ns()
  361|             if self._active_challenge is None or (now_ns - self._active_challenge.timestamp_monotonic_ns) > self._active_challenge.max_age_ns:

  387| 
  388|         If called with an active challenge, returns a canonical structured token string

  391|         P1-1: same authority gate as create_reset_token.

  395|                 raise SafetyError(

  398|                     code="ESTOP_MINT_DENIED",

  402|             # B11: when a live challenge exists, create_reset_token() already

  438|     def trigger(
  439|         self,
  440|         trigger: EStopTriggerSource,
  441|         reason: str,
  442|         vehicle_speed_kmh: float = 0.0,
  443|     ) -> None:
  444|         """Engage Emergency Stop immediately and cut all transmissions."""
  445|         with self._lock:
  446|             now_wall_ns = time.time_ns()
  447|             now_monotonic_ns = time.monotonic_ns()
  448| 
  449|             already_engaged = self._is_engaged
  450|             self._is_engaged = True
  451| 
  452|             # If not already engaged or challenge is missing/expired, issue fresh challenge
  453|             if (
  454|                 not already_engaged
  455|                 or self._active_challenge is None
  456|                 or (now_monotonic_ns - self._active_challenge.timestamp_monotonic_ns) > self._active_challenge.max_age_ns
  457|             ):
  458|                 challenge_nonce = os.urandom(16)
  459|                 self._active_challenge = EStopChallenge(
  460|                     epoch=self._epoch,
  461|                     nonce=challenge_nonce,
  462|                     timestamp_monotonic_ns=now_monotonic_ns,
  463|                     timestamp_wall_ns=now_wall_ns,
  464|                     action="ESTOP_RESET",
  465|                     max_age_ns=self._max_token_age_ns,
  466|                 )
  467|             self._last_event = EStopEvent(
  468|                 trigger=trigger,
  469|                 reason=reason,
  470|                 timestamp_ns=now_wall_ns,
  471|                 system_speed_kmh=vehicle_speed_kmh,
  472|             )
  473|             # CRITICAL-1: every engagement invalidates all in-flight validated
  474|             # frames. The fence bump stays under the estop _lock and MUST NOT
  475|             # take tx_send_lock: a dispatch may block holding tx_send_lock
  476|             # inside driver I/O (see gateway test
  477|             # test_estop_callback_does_not_block_on_slow_driver_io) — if
  478|             # trigger() waited on tx_send_lock, the engagement itself would
  479|             # deadlock behind the in-flight send and the E-Stop callback
  480|             # would never fire. Coordination with the send path is via (a)
  481|             # the PHASE-3 fenced re-check in the gateway (any COMPLETED
  482|             # transition bumps the generation and kills validated frames
  483|             # still waiting for the fence) and (b) abort/flush hooks below
  484|             # that request HAL queue cancellation for the already-dispatched
  485|             # frame. Residual window (engagement landing between fence check
  486|             # and bus write) is documented at the gateway PHASE-3 site.
  487|             self._tx_fence += 1
  488|             event_snapshot = self._last_event
  489|             callbacks_snapshot = list(self._callbacks)
  490|             abort_snapshot = list(self._abort_hooks)
  491| 
  492|         logger.critical(
  493|             "EMERGENCY STOP ENGAGED",
  494|             extra={
  495|                 "trigger": trigger.value,
  496|                 "reason": reason,
  497|                 "speed_kmh": vehicle_speed_kmh,
  498|                 "epoch": self._epoch,
  499|             },
  500|         )
  501| 
  502|         # Notify all listeners outside lock with exception isolation
  503|         for cb in callbacks_snapshot:
  504|             try:
  505|                 cb(event_snapshot)
  506|             except Exception as exc:
  507|                 logger.error("Error in E-Stop callback", extra={"error": str(exc)})
  508|         # Driver abort/flush hooks: request HAL queue cancellation outside locks.
  509|         for hook in abort_snapshot:
  510|             try:
  511|                 hook()
  512|             except Exception as exc:
  513|                 logger.error("Error in E-Stop abort hook", extra={"error": str(exc)})

  515|     def _reset_failure(self, reason: str, sig_prefix: str = "") -> SafetyError:
  516|         """Record a failed reset: counter + exponential backoff + audit log.
  517| 
  518|         The first consecutive failure does NOT arm the cooldown (tolerates a
  519|         single operator typo / stale-token retry); persistence (>=2) is
  520|         throttled with exponential backoff capped at RESET_BACKOFF_CAP_SEC.
  521|         Sustained online guessing is still rate-limited; a lone mistake is not
  522|         punished with a lockout.
  523|         """
  524|         self._failed_reset_attempts += 1
  525|         backoff_sec = 0.0
  526|         if self._failed_reset_attempts >= 2:
  527|             # Cap the exponent BEFORE shifting: 0.2 * 2**5 = 6.4 > 5.0 cap, so
  528|             # shifts beyond 5 never change the result but would overflow float
  529|             # (and int->float conversion) under hammering.
  530|             shift = min(self._failed_reset_attempts - 2, 5)
  531|             backoff_sec = min(RESET_BACKOFF_BASE_SEC * (2**shift), RESET_BACKOFF_CAP_SEC)
  532|             self._reset_backoff_until_ns = time.monotonic_ns() + int(backoff_sec * 1_000_000_000)
  533|         level = "CRITICAL" if self._failed_reset_attempts >= 5 else "WARNING"
  534|         if level == "CRITICAL":
  535|             logger.critical(

  272|                 # Illegal transition: mutate state and snapshot listeners under
  273|                 # the lock, but defer dispatch AND the raise to outside (B6) so
  274|                 # re-entrant callbacks cannot deadlock and no SafetyError ever
  275|                 # unwinds through a held lock.
  276|                 err_msg = (
  277|                     f"Illegal Safety State transition from '{self._state.value}' to '{new_state.value}'. "
  278|                     f"Reason: {reason or 'N/A'}"
  279|                 )
  280|                 logger.critical(err_msg)
  281|                 old_state = self._state
  282|                 self._state = SafetyState.FAULT
  283|                 fault_reason = "ILLEGAL_STATE_TRANSITION: " + err_msg
  284|                 self._fault_reason = fault_reason
  285|                 self._epoch += 1
  286|                 self._state_change_timestamp_ns = now_monotonic_ns
  287|                 self._last_duration_ns = 0
  288| 
  289|                 record = StateTransitionRecord(
  290|                     from_state=old_state,
  291|                     to_state=SafetyState.FAULT,
  292|                     reason=fault_reason,
  293|                     epoch=self._epoch,
  294|                     monotonic_timestamp_ns=now_monotonic_ns,
  295|                     wall_time_utc=now_utc,
  296|                     duration_ns=duration_ns,
  297|                 )
  298|                 self._history.append(record)
  299|                 callbacks_snapshot = list(self._callbacks)
  300|                 illegal = err_msg
  301|             else:
  302|                 old_state = self._state
  303|                 self._state = new_state
  304|                 self._epoch += 1
  305|                 self._state_change_timestamp_ns = now_monotonic_ns
  306|                 self._last_duration_ns = 0
  307| 
  308|                 if new_state == SafetyState.FAULT:
  309|                     self._fault_reason = reason
  310|                 elif old_state == SafetyState.FAULT and new_state in {SafetyState.PASSIVE, SafetyState.SAFE}:
  311|                     self._fault_reason = ""
  312| 
  313|                 record = StateTransitionRecord(
  314|                     from_state=old_state,
  315|                     to_state=new_state,
  316|                     reason=reason,
  317|                     epoch=self._epoch,
  318|                     monotonic_timestamp_ns=now_monotonic_ns,
  319|                     wall_time_utc=now_utc,
  320|                     duration_ns=duration_ns,
  321|                 )
  322|                 self._history.append(record)
  323|                 callbacks_snapshot = list(self._callbacks)
  324| 
  325|         if illegal is not None:
  326|             # Snapshot-Then-Release: dispatch + raise outside the lock (B6)
  327|             self._dispatch_callbacks(callbacks_snapshot, old_state, SafetyState.FAULT, self._fault_reason)
  328|             raise SafetyError(illegal, code="ILLEGAL_SAFETY_TRANSITION")
  329| 
  330|         logger.info(
  331|             "Safety State Transition",
  332|             extra={
  333|                 "from": old_state.value,
  334|                 "to": new_state.value,
  335|                 "reason": reason,
  336|                 "epoch": self._epoch,
  337|                 "duration_ms": duration_ns / 1_000_000.0,
  338|             },
  339|         )
  340| 
  341|         # Snapshot-Then-Release: Execute callbacks outside lock with exception isolation
  342|         self._dispatch_callbacks(callbacks_snapshot, old_state, new_state, reason)
  343| 
  344|     def enter_passive_mode(self, reason: str = "Operator entered PASSIVE listen-only mode") -> None:
  345|         """Safely transition to PASSIVE (Listen-Only) mode."""
  346|         self.transition_to(SafetyState.PASSIVE, reason=reason)

  348|     def issue_arm_token(self, ttl_s: float = 30.0) -> str:
  349|         """Mint a single-use HMAC authorization token for arm_tx/activate_tx.
  350| 
  351|         Requires a configured `auth_secret` (see __init__). The token binds a
  352|         random nonce + monotonic expiry under the secret; it is verified with
  353|         hmac.compare_digest and burned on first use, so a replayed token fails
  354|         closed. Format: ``<expiry_ns>.<nonce_hex>.<mac_hex>``.
  355|         """
  356|         if self._auth_secret is None:
  357|             raise SafetyError(
  358|                 "Arm tokens require a configured supervisor auth_secret",
  359|                 code="ARM_AUTH_NOT_CONFIGURED",
  360|             )
  361|         expiry_ns = time.monotonic_ns() + int(max(1.0, ttl_s) * 1_000_000_000)
  362|         nonce = _secrets.token_bytes(16)
  363|         payload = expiry_ns.to_bytes(8, "big", signed=False) + nonce
  364|         mac = hmac.new(self._auth_secret, payload, hashlib.sha256).digest()
  365|         return f"{expiry_ns}.{nonce.hex()}.{mac.hex()}"
  366| 
  367|     def _verify_arm_token(self, auth_token: str) -> None:
  368|         """Verify a presented arm token (fail-closed: expiry + single-use + HMAC)."""
  369|         assert self._auth_secret is not None
  370|         token = auth_token.strip() if isinstance(auth_token, str) else ""
  371|         parts = token.split(".")
  372|         if len(parts) != 3:
  373|             raise SafetyError(
  374|                 "TX authorization rejected: malformed auth token",
  375|                 code="ARM_AUTH_INVALID",
  376|             )
  377|         expiry_s, nonce_hex, mac_hex = parts
  378|         try:
  379|             expiry_ns = int(expiry_s)
  380|             nonce = bytes.fromhex(nonce_hex)
  381|             mac = bytes.fromhex(mac_hex)
  382|         except ValueError as exc:
  383|             raise SafetyError(
  384|                 "TX authorization rejected: malformed auth token",
  385|                 code="ARM_AUTH_INVALID",
  386|             ) from exc
  387|         payload = expiry_ns.to_bytes(8, "big", signed=False) + nonce
  388|         expected = hmac.new(self._auth_secret, payload, hashlib.sha256).digest()
  389|         if not hmac.compare_digest(mac, expected):
  390|             raise SafetyError(
  391|                 "TX authorization rejected: invalid auth token",
  392|                 code="ARM_AUTH_INVALID",
  393|             )
  394|         if time.monotonic_ns() > expiry_ns:
  395|             raise SafetyError(
  396|                 "TX authorization rejected: auth token expired",
  397|                 code="ARM_AUTH_EXPIRED",
  398|             )
  399|         if payload in self._consumed_arm_tokens:
  400|             raise SafetyError(
  401|                 "TX authorization rejected: auth token already consumed",
  402|                 code="ARM_AUTH_REPLAYED",
  403|             )
  404|         self._consumed_arm_tokens.append(payload)
  405| 
  406|     def _require_arm_authorization(self, operation: str, auth_token: str | None) -> None:
  407|         """Enforce the TX-authorization gate for arm_tx/activate_tx (T41 / S-1).
  408| 
  409|         Fail-closed policy:
  410|           * ``auth_secret`` configured -> a valid, unexpired, single-use token
  411|             is REQUIRED (missing/invalid => SafetyError). The legacy
  412|             ``allow_unauthenticated_arm`` opt-in downgrades this to a WARNING.
  413|           * ``auth_secret`` None -> no authenticator is wired; the legacy
  414|             unauthenticated path is preserved with a WARNING (backward compat).
  415|         """
  416|         if self._auth_secret is not None:
  417|             if auth_token is None:
  418|                 if self._allow_unauthenticated_arm:
  419|                     logger.warning(
  420|                         "%s without auth_token — allow_unauthenticated_arm=True "
  421|                         "override in effect (TEST-ONLY, TX not operator-authenticated)",
  422|                         operation,
  423|                     )
  424|                     return
  425|                 logger.critical(
  426|                     "%s without auth_token while auth_secret is configured — FAIL CLOSED",
  427|                     operation,
  428|                 )
  429|                 raise SafetyError(
  430|                     f"{operation} rejected: auth_token is required "
  431|                     "(a supervisor auth_secret is configured)",
  432|                     code="ARM_AUTH_REQUIRED",
  433|                 )
  434|             self._verify_arm_token(auth_token)
  435|             return
  436|         logger.warning(
  437|             "%s without auth_secret configured (legacy unauthenticated path)",
  438|             operation,
  439|         )

  441|     def arm_tx(
  442|         self,
  443|         reason: str = "Operator explicitly ARMED TX pipeline",
  444|         auth_token: str | None = None,
  445|     ) -> None:
  446|         """Explicitly arm the TX pipeline.
  447| 
  448|         The authorization gate is enforced by ``_require_arm_authorization``:
  449|         when an ``auth_secret`` is configured on this supervisor, ``auth_token``
  450|         is mandatory and HMAC-verified (fail-closed). See the constructor for
  451|         the ``allow_unauthenticated_arm`` compatibility flag.
  452|         """
  453|         self._require_arm_authorization("arm_tx", auth_token)
  454|         self.transition_to(SafetyState.ARMED_TX, reason=reason)

  457|         self,
  458|         reason: str = "TX active transmission ongoing",
  459|         auth_token: str | None = None,
  460|     ) -> None:

  462|         self._require_arm_authorization("activate_tx", auth_token)

  465|     def trigger_fault(self, reason: str = "Safety fault detected") -> None:
  466|         """Trigger FAULT state, immediately revoking all TX authorization.
  467| 
  468|         Rate-limited to FAULT_RATE_LIMIT_PER_SEC per second (fail-closed audit;
  469|         excess calls are logged and dropped to prevent DoS via fault storms).
  470|         """
  471|         now_ns = time.monotonic_ns()
  472|         with self._lock:
  473|             # Prune timestamps older than 1s.
  474|             while self._fault_timestamps_ns and (now_ns - self._fault_timestamps_ns[0]) >= 1_000_000_000:
  475|                 self._fault_timestamps_ns.popleft()
  476|             if len(self._fault_timestamps_ns) >= self.FAULT_RATE_LIMIT_PER_SEC:
  477|                 logger.warning(
  478|                     "trigger_fault rate-limited (fault storm suppressed)",
  479|                     extra={"reason": _sanitize_reason(reason)},
  480|                 )
  481|                 return
  482|             self._fault_timestamps_ns.append(now_ns)
  483|         logger.warning("Safety FAULT triggered", extra={"reason": _sanitize_reason(reason)})
  484|         self.transition_to(SafetyState.FAULT, reason=reason)

  486|     def _force_fault(self, reason: str) -> None:
  487|         """Internal emergency transition to FAULT bypassing transition table.
  488| 
  489|         Adheres strictly to Snapshot-Then-Release locking and complete callback exception isolation.
  490|         """
  491|         reason = _sanitize_reason(reason)
  492|         with self._lock:
  493|             if self._state == SafetyState.FAULT and self._fault_reason == reason:
  494|                 return
  495|             now_monotonic_ns = time.monotonic_ns()
  496|             now_utc = datetime.now(timezone.utc)
  497|             duration_ns = max(0, now_monotonic_ns - self._state_change_timestamp_ns)
  498| 
  499|             old_state = self._state
  500|             self._state = SafetyState.FAULT
  501|             self._fault_reason = reason
  502|             self._epoch += 1
  503|             self._state_change_timestamp_ns = now_monotonic_ns
  504|             self._last_duration_ns = 0
  505| 
  506|             record = StateTransitionRecord(
  507|                 from_state=old_state,
  508|                 to_state=SafetyState.FAULT,
  509|                 reason=reason,
  510|                 epoch=self._epoch,
  511|                 monotonic_timestamp_ns=now_monotonic_ns,
  512|                 wall_time_utc=now_utc,
  513|                 duration_ns=duration_ns,
  514|             )
  515|             self._history.append(record)
  516|             callbacks_snapshot = list(self._callbacks)
  517| 
  518|         logger.critical(
  519|             "Safety State Forced FAULT",
  520|             extra={
  521|                 "from": old_state.value,
  522|                 "to": SafetyState.FAULT.value,
  523|                 "reason": reason,
  524|                 "epoch": self._epoch,
  525|             },
  526|         )
  527| 
  528|         # Snapshot-Then-Release: Execute callbacks outside lock with exception isolation
  529|         self._dispatch_callbacks(callbacks_snapshot, old_state, SafetyState.FAULT, reason)

  531|     @staticmethod
  532|     def _dispatch_callbacks(
  533|         callbacks: list[Callable[[SafetyState, SafetyState, str], None]],
  534|         old_state: SafetyState,
  535|         new_state: SafetyState,
  536|         reason: str,
  537|     ) -> None:
  538|         """Dispatch callbacks outside the lock with complete exception isolation."""
  539|         for cb in callbacks:
  540|             try:
  541|                 cb(old_state, new_state, reason)
  542|             except Exception as exc:
  543|                 logger.error(
  544|                     "Error in Safety State Machine callback",
  545|                     extra={
  546|                         "error": str(exc),
  547|                         "callback": getattr(cb, "__name__", repr(cb)),
  548|                         "from_state": old_state.value,
  549|                         "to_state": new_state.value,
  550|                     },
  551|                     exc_info=True,
  552|                 )
