  305| # Weighted evidence kinds for local-expert root-cause confidence, mirroring
  306| # engine/discovery/evidence.py methodology (P0 honesty fix: replaces the old
  307| # hardcoded "%94 Belirlenimsiz Güvenilirlik" placeholder). Rule scenario and
  308| # knowledge-base matches are complementary identification evidence — a DTC
  309| # explained by a rule does not also require a KB hit.
  310| ROOT_CAUSE_EVIDENCE_WEIGHTS: dict[str, float] = {
  311|     "identification": 0.50,
  312|     "telemetry_correlation": 0.30,
  313|     "dtc_context": 0.20,
  314| }

  316| 
  317| def compute_root_cause_confidence(
  318|     dtc_count: int,
  319|     scenario_matched: int,
  320|     kb_matched: int,
  321|     telemetry_correlation_count: int,
  322| ) -> str:
  323|     """Compute an honest weighted root-cause confidence label for the offline expert.
  324| 
  325|     Each evidence kind is normalised to 0..1, weighted, and the weighted mean
  326|     is taken over the total weight — so a full rule+telemetry match scores
  327|     high while an unknown DTC with no corroboration scores near zero.
  328|     """
  329|     if dtc_count <= 0:
  330|         return "Normal"
  331|     identified = scenario_matched + kb_matched
  332|     entries = {
  333|         "identification": min(1.0, identified / dtc_count),
  334|         "telemetry_correlation": min(1.0, telemetry_correlation_count / 2.0),
  335|         "dtc_context": min(1.0, dtc_count / 2.0),
  336|     }
  337|     total_weight = sum(ROOT_CAUSE_EVIDENCE_WEIGHTS.values())
  338|     weighted_sum = sum(ROOT_CAUSE_EVIDENCE_WEIGHTS[kind] * value for kind, value in entries.items())
  339|     score = weighted_sum / total_weight
  340|     label = "Yüksek" if score >= 0.65 else ("Orta" if score >= 0.35 else "Düşük")
  341|     return f"{label} (%{score * 100:.0f} ağırlıklı kanıt skoru)"

  376| # ISO 15765-2 (ISO-TP) UDS service identifiers recognised by the packet
  377| # explainer — requests and positive/negative responses.
  378| _UDS_KNOWN_SIDS: frozenset[int] = frozenset(
  379|     {
  380|         0x10, 0x11, 0x14, 0x19, 0x22, 0x27, 0x28, 0x2E, 0x31, 0x3E,
  381|         0x50, 0x51, 0x54, 0x59, 0x62, 0x67, 0x71, 0x7F, 0x01,
  382|     }
  383| )

  386| def _resolve_uds_sid_index(payload_bytes: list[int]) -> int:
  387|     """Resolve the byte offset of the UDS SID inside an ISO-TP framed payload.
  388| 
  389|     ISO 15765-2 PCI types (upper nibble of byte 0):
  390|       0x0 = Single Frame (SF)  -> SID at byte 1
  391|       0x1 = First Frame (FF)  -> SID at byte 2 (bytes 0-1 are PCI+length)
  392|       0x2 = Consecutive Frame (CF) -> no reliable SID position
  393|       0x3 = Flow Control (FC) -> not a service payload
  394| 
  395|     P1 fix: the legacy heuristic could misread a First Frame such as
  396|     ``10 14 ...`` as SF with SID 0x14 (ClearDiagnosticInformation). With the
  397|     explicit PCI nibble check, multi-frame requests are offset correctly and
  398|     CF/FC fragments are never mistaken for fresh service requests.
  399|     """
  400|     if not payload_bytes:
  401|         return 0
  402|     pci_type = (payload_bytes[0] >> 4) & 0xF
  403|     if pci_type == 0x0 and len(payload_bytes) > 1:
  404|         return 1  # Single Frame: [0|len, SID, ...]
  405|     if pci_type == 0x1 and len(payload_bytes) > 2:
  406|         return 2  # First Frame: [0x1x, len_hi, len_lo, SID, ...]
  407|     if pci_type in (0x2, 0x3):
  408|         return 0  # CF fragment / FC frame — no reliable SID position
  409|     # Unframed payload (e.g. raw physical request): SID at byte 0 unless
  410|     # byte 0 is not a known SID while byte 1 is.
  411|     if payload_bytes[0] not in _UDS_KNOWN_SIDS and len(payload_bytes) > 1 and payload_bytes[1] in _UDS_KNOWN_SIDS:
  412|         return 1
  413|     return 0

  416| def explain_can_packet(
  417|     can_id_hex_or_int: str | int,
  418|     payload: bytes | list[int] | str = b"",
  419| ) -> tuple[str, list[dict[str, Any]]]:
  420|     """Break down a CAN frame payload into a concise 2-3 line explanation with action triggers."""
  421|     if isinstance(can_id_hex_or_int, str):
  422|         cleaned_id = can_id_hex_or_int.strip().lower()
  423|         if cleaned_id.startswith("0x"):
  424|             can_id = int(cleaned_id, 16)
  425|         else:
  426|             can_id = int(cleaned_id, 16) if all(c in "0123456789abcdef" for c in cleaned_id) else 0
  427|     else:
  428|         can_id = int(can_id_hex_or_int)
  429| 
  430|     if isinstance(payload, str):
  431|         payload_bytes = [int(t, 16) for t in re.findall(r"\b[0-9A-Fa-f]{2}\b", payload)]
  432|     elif isinstance(payload, bytes):
  433|         payload_bytes = list(payload)
  434|     else:
  435|         payload_bytes = list(payload)
  436| 
  437|     # 1. UDS Diagnostics (0x7DF or 0x7E0..0x7EF)
  438|     if (0x7E0 <= can_id <= 0x7EF) or can_id == 0x7DF:
  439|         if payload_bytes:
  440|             sid_idx = _resolve_uds_sid_index(payload_bytes)
  441| 
  442|             sid = payload_bytes[sid_idx]
  443| 
  444|             # 0x10 Diagnostic Session Control
  445|             if sid == 0x10:
  446|                 subfn = payload_bytes[sid_idx + 1] if len(payload_bytes) > sid_idx + 1 else 1
  447|                 sub_names = {1: "Default", 2: "Programming", 3: "Extended", 4: "Safety System"}
  448|                 s_name = sub_names.get(subfn & 0x7F, f"0x{subfn:02X}")

  450|                 line2 = f"• **Servis:** `0x10 DiagnosticSessionControl` — {s_name} Session (0x{subfn:02X})"
  451|                 line3 = f"• **Anlam:** ECU'dan {s_name} oturumuna geçiş talep ediliyor."
  452|                 actions = [make_uds_session_action(subfn & 0x7F)]
  453|                 return (f"{line1}\n{line2}\n{line3}", actions)
  454| 
  455|             # 0x11 ECU Reset

 1383| def _validate_dtc_entry_shape(code: str, info: Any) -> bool:
 1384|     """Shape-validate one external DTC entry before merging (M-17 / P2-16).
 1385| 
 1386|     A hostile or corrupted external DB must not inject garbage into the
 1387|     expert base: `steps` must be a list of 2-3 element sequences of strings
 1388|     (a bare string used to be character-indexed by the consumer), `severity`
 1389|     must be a known level, and the required top-level fields must be present.
 1390|     """
 1391|     if not isinstance(info, dict):
 1392|         return False
 1393|     for required_field in ("title", "subsystem", "severity"):
 1394|         val = info.get(required_field)
 1395|         if not isinstance(val, str) or not val.strip():
 1396|             return False
 1397|     severity = info.get("severity")
 1398|     if severity not in ("INFO", "LOW", "MEDIUM", "HIGH", "CRITICAL", "CRITICAL_STOP"):
 1399|         return False
 1400|     steps = info.get("steps")
 1401|     if steps is not None:
 1402|         if not isinstance(steps, (list, tuple)):
 1403|             return False
 1404|         for s in steps:
 1405|             if not isinstance(s, (list, tuple)) or not 2 <= len(s) <= 3:
 1406|                 return False
 1407|             if not all(isinstance(part, str) for part in s):
 1408|                 return False
 1409|     causes = info.get("causes")
 1410|     if causes is not None and not isinstance(causes, (list, tuple)):
 1411|         return False
 1412|     _ = code  # validated by caller (dict key, always str from JSON)
 1413|     return True

 1416| def load_external_dtc_database(data_path: Path | str | None = None) -> int:
 1417|     """Dynamically load and merge external DTC catalog into EXPERT_KNOWLEDGE_BASE.
 1418| 
 1419|     Preserves hardcoded expert rules (existing entries are never overwritten).
 1420|     Returns the number of new DTC codes merged.
 1421|     """
 1422|     target = Path(data_path) if data_path else _EXTERNAL_DATA_DIR / "dtc_database.json"
 1423|     if not target.exists():
 1424|         # H-9 (P1-11): missing DB is a DEGRADED mode, not business as usual —
 1425|         # debug level hid the loss of ~98% of the knowledge base in frozen
 1426|         # builds. Surface it at WARNING so operators can notice.
 1427|         logger.warning("External DTC database not found at %s — knowledge base degraded to built-in rules", target)
 1428|         return 0
 1429| 
 1430|     try:
 1431|         content = target.read_text(encoding="utf-8", errors="replace")
 1432|         data = json.loads(content)
 1433|         if not isinstance(data, dict):
 1434|             logger.warning("External DTC database format invalid (expected dict, got %s)", type(data).__name__)
 1435|             return 0
 1436| 
 1437|         added = 0
 1438|         rejected = 0
 1439|         for code, info in data.items():
 1440|             # M-17 (P2-16): shape validation BEFORE merge — garbage entries
 1441|             # are counted and skipped, never merged.
 1442|             if not _validate_dtc_entry_shape(code, info):
 1443|                 rejected += 1
 1444|                 continue
 1445|             # Preserve existing rich hand-crafted rules
 1446|             if code not in EXPERT_KNOWLEDGE_BASE:
 1447|                 EXPERT_KNOWLEDGE_BASE[code] = info
 1448|                 added += 1
 1449| 
 1450|         if rejected:
 1451|             logger.warning(
 1452|                 "External DTC database: %d entries rejected by shape validation", rejected
 1453|             )
 1454|         logger.info("Merged %d external DTC codes into EXPERT_KNOWLEDGE_BASE (total: %d)", added, len(EXPERT_KNOWLEDGE_BASE))
 1455|         return added
 1456|     except Exception as exc:
 1457|         logger.warning("Failed to load external DTC database from %s: %s", target, exc)
 1458|         return 0
