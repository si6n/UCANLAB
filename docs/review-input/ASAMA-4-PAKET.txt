  209| class TransportAnomalyMetrics:
  210|     """Observable TP anomaly counters (MEDIUM-7): ABORT/CTS storm detection.
  211| 
  212|     A bus-hijack / DoS attempt shows up on J1939-21 as an abnormal ratio
  213|     of Conn_Abort frames and unsolicited CTS bursts against normal RTS/BAM
  214|     traffic. These counters are pure observability — they never gate RX
  215|     (the session-hardening paths do that); callers poll and raise alarms.
  216|     """
  217| 
  218|     # RX-side TP.CM counters (per control byte)
  219|     rx_abort: int = 0
  220|     rx_rts: int = 0
  221|     rx_bam: int = 0
  222|     rx_cts: int = 0
  223|     # CTS frames addressed to a sender session that does not exist
  224|     # (unsolicited CTS — classic spoof/injection signature)
  225|     rx_unsolicited_cts: int = 0
  226|     # Aborts we emitted in response to detected anomalies
  227|     tx_abort_emitted: int = 0
  228|     # REVIEW 2-H2 (HIGH): J1939-22 FD TP frames (>8 bytes) that the
  229|     # classic J1939-21 engine must drop — surfaced, never silent.
  230|     dropped_fd_tp_frames: int = 0

  232|     @property
  233|     def abort_ratio(self) -> float:
  234|         """Abort share of all received TP.CM connection frames (RTS/CTS/ABORT).
  235| 
  236|         A healthy bus stays near 0; a hijacker spraying Abort frames
  237|         against sessions drives this toward 1.
  238|         """
  239|         total = self.rx_abort + self.rx_rts + self.rx_cts
  240|         if total == 0:
  241|             return 0.0
  242|         return self.rx_abort / total

  244|     def is_suspect(self, *, min_frames: int = 20, ratio_threshold: float = 0.5) -> bool:
  245|         """Heuristic bus-hijack alarm (MEDIUM-7).
  246| 
  247|         True when at least `min_frames` connection-mode TP.CM frames have
  248|         been observed AND the abort ratio exceeds `ratio_threshold`, or
  249|         when any unsolicited CTS has been observed at all. Conservative
  250|         defaults: needs >= 20 frames and > 50% aborts — a normal CMDT
  251|         exchange (RTS -> CTS -> DT -> ACK) never trips this.
  252|         """
  253|         if self.rx_unsolicited_cts > 0:
  254|             return True
  255|         if (self.rx_abort + self.rx_rts + self.rx_cts) < min_frames:

  399|     def handle_rx_frame(self, frame: CanFrame) -> tuple[CompletedMessage | None, CanFrame | None]:
  400|         """Process incoming frame according to SAE J1939-21 PDU format rules."""
  401|         fd_tp = frame.is_fd and (frame.dlc > 8 or len(frame.data) > 8)
  402|         if not frame.is_extended or len(frame.data) < 8 or fd_tp:
  403|             # M-34 (micro) -> REVIEW 2-H2 (HIGH): J1939-22/FD TP frames
  404|             # (>8-byte transport payloads) are NOT decodable by the classic
  405|             # J1939-21 engine. They previously fell into this guard and
  406|             # vanished at DEBUG level. FD TP frames are now counted in
  407|             # anomaly_metrics.dropped_fd_tp_frames and logged at WARNING —
  408|             # a J1939-22 capture must be visible, not silently empty.
  409|             # Non-FD short/11-bit noise stays at DEBUG (hot path).
  410|             if fd_tp:
  411|                 with self._sessions_lock:
  412|                     self.anomaly_metrics.dropped_fd_tp_frames += 1
  413|                 logger.warning(
  414|                     "J1939-22 FD transport frame dropped — classic J1939-21 engine only",
  415|                     extra={
  416|                         "is_fd": frame.is_fd,
  417|                         "dlc": frame.dlc,
  418|                         "data_len": len(frame.data),
  419|                         "arbitration_id": frame.arbitration_id,
  420|                     },
  421|                 )
  422|                 return None, None
  423|             logger.debug(
  424|                 "Dropping non-J1939-21-transport frame (11-bit or short/DL)",
  425|                 extra={
  426|                     "is_extended": frame.is_extended,
  427|                     "data_len": len(frame.data),
  428|                     "arbitration_id": frame.arbitration_id,
  429|                 },
  430|             )
  431|             return None, None
  432| 
  433|         # 29-bit CAN ID decomposition — M-12 (P2-6): shared parser preserves
  434|         # EDP/DP; the old hand-rolled math dropped EDP, mis-deriving the PGN
  435|         # for EDP-set frames and corrupting TP session keying.
  436|         pgn, sa, da_opt, _priority = parse_j1939_id(frame.arbitration_id)
  437|         da = da_opt if da_opt is not None else 255
  438| 
  439|         # Check for TP.CM (PGN 60416 / 0xEC00)
  440|         if pgn == PGN_TP_CM:
  441|             # P2-4: route by SESSION EXISTENCE, not control byte alone. A
  442|             # peer Abort (DA=us) that has no matching TX session is aimed at
  443|             # our RECEIVER session — the old control-byte dispatch dropped
  444|             # it, leaking the RX session until reaping.
  445|             tx_key = (self.my_address, sa, pgn_from_tp_cm(frame), frame.channel_id) if ctrl_targets_sender(frame) else None
  446|             if da == self.my_address and ctrl_targets_sender(frame):
  447|                 # Abort with no TX session → fall through to the RX handler
  448|                 # so the receiving side can drop its session.
  449|                 has_tx_session = False
  450|                 if tx_key is not None:
  451|                     with self._sessions_lock:
  452|                         probe_key = (self.my_address, sa, pgn_from_tp_cm(frame), frame.channel_id)
  453|                         has_tx_session = probe_key in self._tx_sessions
  454|                 if has_tx_session or frame.data[0] != TP_CTRL_ABORT:
  455|                     dt_frames, err_frame = self._handle_tx_cm(frame, sa, da)
  456|                     if dt_frames:
  457|                         # First frame rides the single response slot; the rest
  458|                         # queue. M-6 (3FABLE): the queue mutation is under
  459|                         # _sessions_lock — the RX thread and the
  460|                         # take_pending_tx_frames drain must not interleave.
  461|                         with self._sessions_lock:
  462|                             self._pending_tx_frames.extend(dt_frames[1:])
  463|                             if err_frame is not None:
  464|                                 self._pending_tx_frames.append(err_frame)
  465|                         return None, dt_frames[0]
  466|                     if err_frame is not None:
  467|                         return None, err_frame
  468|                     return None, None
  469|             return self._handle_tp_cm(frame, sa, da)
  470| 
  471|         # Check for TP.DT (PGN 60160 / 0xEB00)
  472|         if pgn == PGN_TP_DT:
  473|             return self._handle_tp_dt(frame, sa, da)
  474| 
  475|         # REVIEW 3-3 (HIGH): ETP.CM / ETP.DT are recognized and surfaced —
  476|         # never silently dropped. Full ETP state machine is not implemented;
  477|         # logging makes an unsupported ETP ECU diagnosable instead of "no data".
  478|         if pgn in (PGN_ETP_CM, PGN_ETP_DT):
  479|             logger.warning(
  480|                 "J1939 ETP frame received but Extended Transport Protocol is not supported",
  481|                 extra={"pgn": pgn, "sa": sa, "da": da, "arbitration_id": frame.arbitration_id},
  482|             )
  483|             return None, None
  484| 
  485|         return None, None

  525|     def _handle_tp_cm(
  526|         self, frame: CanFrame, sa: int, da: int
  527|     ) -> tuple[CompletedMessage | None, CanFrame | None]:
  528|         ctrl_byte = frame.data[0]
  529|         total_bytes = int.from_bytes(frame.data[1:3], byteorder="little")
  530|         total_packets = frame.data[3]
  531|         target_pgn = int.from_bytes(frame.data[5:8], byteorder="little")
  532| 
  533|         # REVIEW hardening: session key scoped by
  534|         # (source_address, destination_address, channel_id, target_pgn).
  535|         # A spoofed RTS with a different PGN no longer aborts a
  536|         # legitimate in-flight session — it opens a parallel entry
  537|         # (quota/capacity still bound). Same-4-tuple RTS keeps the
  538|         # previous abort-and-replace collision path.
  539|         session_key = (sa, da, frame.channel_id, target_pgn)
  540| 
  541|         # Validate SAE J1939-21 TP limits: max 1785 bytes, packets must match declared bytes
  542|         if ctrl_byte in {TP_CTRL_BAM, TP_CTRL_RTS}:
  543|             if not (1 <= total_bytes <= 1785) or total_packets == 0:
  544|                 logger.warning(
  545|                     "Rejected malformed J1939 TP.CM length",
  546|                     extra={"total_bytes": total_bytes, "total_packets": total_packets, "sa": sa},
  547|                 )
  548|                 return None, None
  549|             expected_packets = (total_bytes + 6) // 7
  550|             if total_packets != expected_packets:
  551|                 logger.warning(
  552|                     "Rejected J1939 TP.CM packet count mismatch",
  553|                     extra={"declared": total_packets, "expected": expected_packets, "sa": sa},
  554|                 )
  555|                 return None, None
  556| 
  557|         with self._sessions_lock:
  558|             self._reap_stale_sessions()
  559| 
  560|             if ctrl_byte == TP_CTRL_BAM:
  561|                 self.anomaly_metrics.rx_bam += 1
  562|                 # Broadcast Announce Message must be addressed to global broadcast address DA == 255 (0xFF)
  563|                 if da != 255:
  564|                     logger.warning(
  565|                         "Rejected J1939 TP.CM_BAM with non-broadcast destination address",
  566|                         extra={"da": da, "sa": sa},
  567|                     )
  568|                     return None, None
  569|                 # B-03: If an in-flight BAM is replaced by another BAM from same SA for a DIFFERENT PGN,
  570|                 # record partial drop and log warning cleanly.
  571|                 # REVIEW hardening: same-PGN duplicates keep the legacy
  572|                 # replace path unconditionally. A cross-PGN BAM against a
  573|                 # live prefix is rate-gated — a slow/occasional replacement
  574|                 # keeps B-03 replace semantics (backward compatible); a
  575|                 # burst of distinct-PGN BAMs is a spoof storm and the
  576|                 # newcomer is rejected (live session protected).
  577|                 old_bam = self._rx_sessions.get(session_key)
  578|                 if old_bam is not None:
  579|                     if old_bam.target_pgn != target_pgn:
  580|                         logger.warning(
  581|                             "New BAM with different PGN replaces in-flight session (B-03)",
  582|                             extra={"sa": sa, "old_pgn": hex(old_bam.target_pgn), "new_pgn": hex(target_pgn)},
  583|                         )
  584|                     else:
  585|                         logger.warning(
  586|                             "New BAM replaces in-flight session",
  587|                             extra={"sa": sa, "old_pgn": hex(old_bam.target_pgn), "new_pgn": hex(target_pgn)},
  588|                         )
  589|                     self._release_session_slot(session_key, old_bam)
  590|                 else:
  591|                     live_prefix = [
  592|                         (k, s)
  593|                         for k, s in self._rx_sessions.items()
  594|                         if isinstance(k, tuple)
  595|                         and len(k) == 4
  596|                         and k[0] == sa
  597|                         and k[1] == da
  598|                         and k[2] == frame.channel_id
  599|                     ]
  600|                     if live_prefix:
  601|                         prefix_key = (sa, da, frame.channel_id)
  602|                         if not self._check_prefix_collision_rate(prefix_key):
  603|                             logger.warning(
  604|                                 "J1939 cross-PGN BAM storm — live session protected",
  605|                                 extra={
  606|                                     "sa": sa,
  607|                                     "live_pgn": hex(live_prefix[0][1].target_pgn),
  608|                                     "new_pgn": hex(target_pgn),
  609|                                 },
  610|                             )
  611|                             return None, None
  612|                         live_key, live_sess = live_prefix[0]
  613|                         logger.warning(
  614|                             "New BAM with different PGN replaces in-flight session (B-03)",
  615|                             extra={"sa": sa, "old_pgn": hex(live_sess.target_pgn), "new_pgn": hex(target_pgn)},
  616|                         )
  617|                         self._release_session_slot(live_key, live_sess)
  618| 
  619|                 # F-19: per source-address quota
  620|                 if (
  621|                     session_key not in self._rx_sessions
  622|                     and self._per_sa_sessions[str(sa)] >= self.MAX_SESSIONS_PER_SA
  623|                 ):
  624|                     logger.warning(
  625|                         "Rejected BAM: per-source session quota exceeded",
  626|                         extra={"sa": sa, "quota": self.MAX_SESSIONS_PER_SA},
  627|                     )
  628|                     return None, None
  629| 
  630|                 if len(self._rx_sessions) >= self.MAX_CONCURRENT_SESSIONS and session_key not in self._rx_sessions:
  631|                     oldest_key = min(self._rx_sessions.keys(), key=lambda k: self._rx_sessions[k].last_activity_time)
  632|                     self._release_session_slot(oldest_key, self._rx_sessions[oldest_key])
  633| 
  634|                 logger.debug(
  635|                     "Received J1939 TP.CM_BAM",
  636|                     extra={"sa": sa, "target_pgn": hex(target_pgn), "bytes": total_bytes, "packets": total_packets},
  637|                 )
  638|                 self._rx_sessions[session_key] = ReassemblySession(
  639|                     source_address=sa,
  640|                     destination_address=da,
  641|                     target_pgn=target_pgn,
  642|                     total_bytes=total_bytes,
  643|                     total_packets=total_packets,
  644|                     is_bam=True,
  645|                     expected_sequence=1,
  646|                     last_activity_time=self._get_now(),
  647|                     channel_id=frame.channel_id,
  648|                 )
  649|                 self._per_sa_sessions[str(sa)] += 1
  650|                 return None, None
  651| 
  652|         if ctrl_byte == TP_CTRL_RTS:
  653|             self.anomaly_metrics.rx_rts += 1
  654|             # Request To Send (Point-to-Point CMDT)
  655|             # 1. Reject RTS addressed to global broadcast address DA == 255 (0xFF)
  656|             if da == 255 or da != self.my_address:
  657|                 logger.warning(
  658|                     "Rejected J1939 TP.CM_RTS with invalid destination address",
  659|                     extra={"da": da, "my_address": self.my_address, "sa": sa},
  660|                 )
  661|                 return None, None
  662| 
  663|             # P7: the entire RTS session-mutation path runs under the sessions
  664|             # lock — collision check, quota, capacity and slot creation.
  665|             with self._sessions_lock:
  666|                 # 2. Check for active session collision on
  667|                 # (SA, DA, channel, PGN).
  668|                 # REVIEW hardening: same-PGN duplicates keep the legacy
  669|                 # abort-and-replace path. A cross-PGN RTS against an
  670|                 # ACTIVE prefix is rate-gated: an occasional (slow)
  671|                 # cross-PGN RTS keeps abort-and-replace (backward
  672|                 # compatible with the collision tests); a burst of
  673|                 # distinct-PGN RTS frames is a spoof storm — the live
  674|                 # session is then protected and the newcomer is rejected
  675|                 # with an Abort (live PGN preserved).
  676|                 existing_session = self._rx_sessions.get(session_key)
  677|                 abort_frame: CanFrame | None = None
  678|                 if existing_session is not None:
  679|                     logger.warning(
  680|                         "J1939 TP session collision detected on (SA, DA, channel, PGN)",
  681|                         extra={
  682|                             "sa": sa,
  683|                             "da": da,
  684|                             "old_pgn": hex(existing_session.target_pgn),
  685|                             "new_pgn": hex(target_pgn),
  686|                         },
  687|                     )
  688|                     # Abort existing session with reason=2 (Session Collision)
  689|                     abort_frame = self._create_abort_frame(
  690|                         existing_session, reason=ABORT_REASON_SESSION_COLLISION
  691|                     )
  692|                     self._release_session_slot(session_key, existing_session)
  693|                 else:
  694|                     # No same-PGN entry — look for a live cross-PGN prefix.
  695|                     prefix_key = (sa, da, frame.channel_id)
  696|                     live_prefix = [
  697|                         (k, s)
  698|                         for k, s in self._rx_sessions.items()
  699|                         if isinstance(k, tuple)
  700|                         and len(k) == 4
  701|                         and k[0] == sa
  702|                         and k[1] == da
  703|                         and k[2] == frame.channel_id
  704|                     ]
  705|                     if live_prefix:
  706|                         if not self._check_prefix_collision_rate(prefix_key):
  707|                             # Spoof storm: protect the live session, abort
  708|                             # the newcomer (its PGN survives in the frame).
  709|                             live_key, live_sess = live_prefix[0]
  710|                             logger.warning(
  711|                                 "J1939 cross-PGN RTS storm — live session protected",
  712|                                 extra={
  713|                                     "sa": sa,
  714|                                     "da": da,
  715|                                     "live_pgn": hex(live_sess.target_pgn),
  716|                                     "new_pgn": hex(target_pgn),
  717|                                 },
  718|                             )
  719|                             storm_abort = self._create_abort_frame(
  720|                                 ReassemblySession(
  721|                                     source_address=sa,
  722|                                     destination_address=da,
  723|                                     target_pgn=target_pgn,
  724|                                     total_bytes=total_bytes,
  725|                                     total_packets=total_packets,
  726|                                     is_bam=False,
  727|                                     channel_id=frame.channel_id,
  728|                                 ),
  729|                                 reason=ABORT_REASON_SESSION_COLLISION,
  730|                             )
  731|                             return None, storm_abort
  732|                         # Slow/occasional cross-PGN RTS: legacy
  733|                         # abort-and-replace of the live prefix entry.
  734|                         live_key, live_sess = live_prefix[0]
  735|                         logger.warning(
  736|                             "J1939 TP session collision detected on (SA, DA, channel) cross-PGN",
  737|                             extra={
  738|                                 "sa": sa,
  739|                                 "da": da,
  740|                                 "old_pgn": hex(live_sess.target_pgn),
  741|                                 "new_pgn": hex(target_pgn),
  742|                             },
  743|                         )
  744|                         abort_frame = self._create_abort_frame(
  745|                             live_sess, reason=ABORT_REASON_SESSION_COLLISION
  746|                         )
  747|                         self._release_session_slot(live_key, live_sess)
  748| 
  749|                 # F-19: per source-address quota (RTS path)
  750|                 if (
  751|                     session_key not in self._rx_sessions
  752|                     and self._per_sa_sessions[str(sa)] >= self.MAX_SESSIONS_PER_SA
  753|                 ):
  754|                     logger.warning(
  755|                         "Rejected RTS: per-source session quota exceeded",
  756|                         extra={"sa": sa, "quota": self.MAX_SESSIONS_PER_SA},
  757|                     )
  758|                     return None, abort_frame
  759| 
  760|                 # Capacity management
  761|                 if len(self._rx_sessions) >= self.MAX_CONCURRENT_SESSIONS and session_key not in self._rx_sessions:
  762|                     oldest_key = min(self._rx_sessions.keys(), key=lambda k: self._rx_sessions[k].last_activity_time)
  763|                     self._release_session_slot(oldest_key, self._rx_sessions[oldest_key])
  764| 
  765|                 rx_win = self.rx_cts_window or getattr(self, "RX_CTS_WINDOW", 0)
  766|                 max_packets_cts = frame.data[4] if len(frame.data) >= 5 and frame.data[4] > 0 else 0xFF
  767|                 # Establish new session with new expected_pgn
  768|                 new_session = ReassemblySession(
  769|                     source_address=sa,
  770|                     destination_address=da,
  771|                     target_pgn=target_pgn,
  772|                     total_bytes=total_bytes,
  773|                     total_packets=total_packets,
  774|                     is_bam=False,
  775|                     expected_sequence=1,
  776|                     last_activity_time=self._get_now(),
  777|                     channel_id=frame.channel_id,
  778|                     rx_cts_window=rx_win,
  779|                     max_packets_per_cts=max_packets_cts,
  780|                 )
  781|                 self._rx_sessions[session_key] = new_session
  782|                 self._per_sa_sessions[str(sa)] += 1
  783| 
  784|                 # If collision occurred, emit abort frame for the old session and
  785|                 # queue the new session's CTS so the peer is granted the transfer
  786|                 # (single response slot cannot carry both frames).
  787|                 if abort_frame is not None:
  788|                     self._pending_tx_frames.append(self._create_cts_frame(new_session))
  789|                     return None, abort_frame
  790| 
  791|                 # Otherwise emit affirmative TP.CM_CTS for the new session
  792|                 cts_frame = self._create_cts_frame(new_session)
  793|                 return None, cts_frame
  794| 
  795|         if ctrl_byte == TP_CTRL_ABORT:
  796|             self.anomaly_metrics.rx_abort += 1
  797|             # Peer Connection Abort (P7: slot release under the sessions lock)
  798|             with self._sessions_lock:
  799|                 logger.warning(
  800|                     "Received J1939 TP.Conn_Abort",
  801|                     extra={"sa": sa, "da": da, "target_pgn": hex(target_pgn), "reason": frame.data[1]},
  802|                 )
  803|                 self._release_session_slot(session_key, self._rx_sessions.get(session_key))
  804|             return None, None
  805| 
  806|         return None, None

  808|     # Cross-PGN collision storm gate: at most this many fast cross-PGN
  809|     # RTS frames per window per (SA, DA, channel) before the live session
  810|     # is protected instead of replaced.
  811|     PREFIX_COLLISION_BURST: int = 3
  812|     PREFIX_COLLISION_WINDOW_S: float = 1.0
  813|     # REVIEW hardening: legacy (SA, DA, channel) compatibility view.
  814|     # _rx_sessions is PGN-scoped internally; this helper exposes the
  815|     # legacy 3-tuple projection for diagnostics/tests that predate the
  816|     # PGN-scoped key (read-only snapshot, no mutation).
  817| 
  818|     def _check_prefix_collision_rate(self, prefix_key: tuple[int, int, str]) -> bool:
  819|         """Cross-PGN collision rate gate. Caller holds _sessions_lock.
  820| 
  821|         Returns True when a legacy abort-and-replace may proceed (slow /
  822|         occasional collision), False when the prefix is under a spoof
  823|         storm (protect the live session).
  824|         """
  825|         now = self._get_now()
  826|         mark = self._rts_prefix_marks.get(prefix_key)
  827|         if mark is None:
  828|             self._rts_prefix_marks[prefix_key] = [now, 1.0]
  829|             return True
  830|         window_start, count = mark
  831|         if (now - window_start) > self.PREFIX_COLLISION_WINDOW_S:
  832|             mark[0] = now
  833|             mark[1] = 1.0
  834|             return True
  835|         if count >= self.PREFIX_COLLISION_BURST:
  836|             return False
  837|         mark[1] = count + 1.0
  838|         return True

  840|     def _release_session_slot(
  841|         self, key: tuple, session: ReassemblySession | None
  842|     ) -> None:
  843|         """Remove a session and decrement its per-SA quota slot.
  844| 
  845|         REVIEW hardening: accepts both 4-tuple (SA, DA, channel, PGN) and
  846|         legacy 3-tuple keys (resolved via the PGN-scoped table).
  847|         """
  848|         if session is not None:
  849|             full_key = (session.source_address, session.destination_address, session.channel_id, session.target_pgn)
  850|             removed = self._rx_sessions.pop(full_key, None)
  851|             if removed is None:
  852|                 removed = self._rx_sessions.pop(key, None)
  853|             if removed is not None:
  854|                 sa_key = str(session.source_address)
  855|                 if self._per_sa_sessions[sa_key] <= 1:
  856|                     self._per_sa_sessions.pop(sa_key, None)
  857|                 else:
  858|                     self._per_sa_sessions[sa_key] -= 1
  859|             return
  860|         self._rx_sessions.pop(key, None)

  862|     def _lookup_dt_session(self, sa: int, da: int, channel_id: str) -> tuple[tuple | None, ReassemblySession | None]:
  863|         """Resolve the RX session owning a TP.DT frame.
  864| 
  865|         TP.DT carries no PGN, so the 4-tuple key cannot be built directly.
  866|         Unique match wins; ambiguous (parallel PGN) DT is dropped
  867|         fail-closed (no cross-PGN payload mixing).
  868|         """
  869|         candidates = [
  870|             (key, sess)
  871|             for key, sess in self._rx_sessions.items()
  872|             if isinstance(key, tuple)
  873|             and len(key) == 4
  874|             and key[0] == sa
  875|             and key[1] == da
  876|             and key[2] == channel_id
  877|         ]
  878|         if len(candidates) == 1:
  879|             return candidates[0]
  880|         if len(candidates) > 1:
  881|             logger.warning(
  882|                 "J1939 TP.DT dropped: ambiguous parallel sessions (fail-closed)",
  883|                 extra={"sa": sa, "da": da, "candidates": len(candidates)},
  884|             )
  885|         return None, None

  887|     def _handle_tp_dt(
  888|         self, frame: CanFrame, sa: int, da: int
  889|     ) -> tuple[CompletedMessage | None, CanFrame | None]:
  890|         seq_num = frame.data[0]
  891|         payload = frame.data[1:8]
  892| 
  893|         # REVIEW hardening: PGN-scoped session lookup for TP.DT (which
  894|         # carries no PGN itself). Unique match wins; ambiguous parallel
  895|         # sessions drop the DT fail-closed.
  896|         # P-C-002 fix: the entire session mutation — sequence check, payload
  897|         # append, completion, and slot release — runs under one lock hold so
  898|         # concurrent TP.DT frames cannot interleave with torn state.
  899|         with self._sessions_lock:
  900|             session_key, session = self._lookup_dt_session(sa, da, frame.channel_id)
  901| 
  902|             if session is None or session_key is None:
  903|                 return None, None
  904| 
  905|             now = self._get_now()
  906| 
  907|             # Session hold: T4 (1050 ms) for CMDT peer-to-peer sessions, T1
  908|             # (750 ms) for BAM broadcasts (P-c — aligned with _reap_stale_sessions)
  909|             hold_timeout = self.T4_TIMEOUT_SEC if not session.is_bam else self.T1_TIMEOUT_SEC
  910|             if (now - session.last_activity_time) > hold_timeout:
  911|                 logger.warning(
  912|                     "J1939 TP.DT session timeout (hold window exceeded)",
  913|                     extra={
  914|                         "sa": sa,
  915|                         "da": da,
  916|                         "target_pgn": hex(session.target_pgn),
  917|                         "timeout_s": hold_timeout,
  918|                         "is_bam": session.is_bam,
  919|                     },
  920|                 )
  921|                 self._release_session_slot(session_key, session)
  922|                 abort_frame = self._create_abort_frame(session, reason=ABORT_REASON_TIMEOUT)
  923|                 return None, abort_frame
  924| 
  925|             # Check sequence order
  926|             if seq_num != session.expected_sequence:
  927|                 logger.warning(
  928|                     "J1939 TP.DT out of order sequence",
  929|                     extra={
  930|                         "expected": session.expected_sequence,
  931|                         "got": seq_num,
  932|                         "sa": sa,
  933|                         "da": da,
  934|                         "target_pgn": hex(session.target_pgn),
  935|                     },
  936|                 )
  937|                 self._release_session_slot(session_key, session)
  938|                 abort_frame = self._create_abort_frame(session, reason=ABORT_REASON_SEQUENCE_ERROR)
  939|                 return None, abort_frame
  940| 
  941|             # Append payload data
  942|             bytes_needed = session.total_bytes - len(session.received_bytes)
  943|             chunk = payload[:bytes_needed]
  944|             session.received_bytes.extend(chunk)
  945|             session.expected_sequence += 1
  946|             session.last_activity_time = now
  947| 
  948|             # Check if transfer is complete
  949|             if session.expected_sequence > session.total_packets or len(session.received_bytes) >= session.total_bytes:
  950|                 completed_data = bytes(session.received_bytes[: session.total_bytes])
  951|                 completed_msg = CompletedMessage(
  952|                     source_address=session.source_address,
  953|                     destination_address=session.destination_address,
  954|                     pgn=session.target_pgn,
  955|                     data=completed_data,
  956|                     timestamp_ns=frame.timestamp_ns,
  957|                     channel_id=frame.channel_id,
  958|                 )
  959| 
  960|                 resp_frame = None
  961|                 if not session.is_bam and session.destination_address == self.my_address:
  962|                     # Send TP.CM_EndOfMsgACK
  963|                     ack_data = bytearray(8)
  964|                     ack_data[0] = TP_CTRL_ACK
  965|                     ack_data[1:3] = session.total_bytes.to_bytes(2, byteorder="little")
  966|                     ack_data[3] = session.total_packets
  967|                     ack_data[4] = 0xFF
  968|                     ack_data[5:8] = session.target_pgn.to_bytes(3, byteorder="little")
  969| 
  970|                     can_id = 0x1CEC0000 | (session.source_address << 8) | (self.my_address & 0xFF)
  971|                     resp_frame = CanFrame.create(
  972|                         channel_id=frame.channel_id,
  973|                         arbitration_id=can_id,
  974|                         data=bytes(ack_data),
  975|                         is_extended=True,
  976|                         direction="tx",
  977|                     )
  978| 
  979|                 self._release_session_slot(session_key, session)
  980|                 return completed_msg, resp_frame
  981| 
  982|             # Partial transfer on a CMDT session: issue the next CTS window so
  983|             # the sender can continue. The receiver's grant policy is bounded by
  984|             # its remaining buffer, expressed via rx_cts_window (0 = grant all).
  985|             # REVIEW hardening: with capped emission (_create_cts_frame),
  986|             # an unconfigured window (0) defaults to CTS-cap-sized
  987|             # follow-ups — otherwise a capped initial grant would stall the
  988|             # transfer forever (self-DoS: no follow-up CTS ever issued).
  989|             if not session.is_bam and session.destination_address == self.my_address:
  990|                 rx_window = (
  991|                     getattr(session, "rx_cts_window", 0)
  992|                     or getattr(self, "RX_CTS_WINDOW", 0)
  993|                     or MAX_CTS_PACKET_COUNT
  994|                 )
  995|                 if rx_window > 0 and (session.expected_sequence - 1) % rx_window == 0:
  996|                     remaining_packets = session.total_packets - (session.expected_sequence - 1)
  997|                     # REVIEW hardening: clamp our own windowed grants to the
  998|                     # CTS cap (see _create_cts_frame).
  999|                     grant = min(rx_window, remaining_packets, MAX_CTS_PACKET_COUNT)
 1000|                     cts_data = bytearray(8)
 1001|                     cts_data[0] = TP_CTRL_CTS
 1002|                     cts_data[1] = grant
 1003|                     cts_data[2] = session.expected_sequence
 1004|                     cts_data[3] = 0xFF
 1005|                     cts_data[4] = 0xFF
 1006|                     cts_data[5:8] = session.target_pgn.to_bytes(3, byteorder="little")
 1007|                     can_id = 0x1CEC0000 | (session.source_address << 8) | (self.my_address & 0xFF)
 1008|                     return None, CanFrame.create(
 1009|                         channel_id=frame.channel_id,
 1010|                         arbitration_id=can_id,
 1011|                         data=bytes(cts_data),
 1012|                         is_extended=True,
 1013|                         direction="tx",
 1014|                     )

 1016|             return None, None

 1018|     def _create_cts_frame(self, session: ReassemblySession) -> CanFrame:
 1019|         """Construct standard J1939 TP.CM_CTS frame (PGN 60416 / 0xEC00 with Control Byte 0x11).
 1020| 
 1021|         REVIEW hardening: our own grants are capped at MAX_CTS_PACKET_COUNT
 1022|         — large transfers proceed in capped windows. A grant-all emission
 1023|         would trip the receiver-side cap (spoof protection) and abort our
 1024|         own legitimate transfers (self-DoS), so emission and receipt caps
 1025|         are symmetric by construction.
 1026|         """
 1027|         grant = session.total_packets
 1028|         if session.max_packets_per_cts > 0 and session.max_packets_per_cts < session.total_packets:
 1029|             grant = min(grant, session.max_packets_per_cts)
 1030|         # REVIEW 1-M1 (MEDIUM): rx_cts_window must bound the FIRST grant too
 1031|         # — previously the initial CTS granted the whole transfer (or the
 1032|         # sender's max) while the DT handler re-issued CTS every rx_window
 1033|         # packets, producing duplicate CTS frames against a fully-granted
 1034|         # sender. Windowed receivers now emit windowed grants from the start;
 1035|         # the default rx_cts_window=0 path is unchanged.
 1036|         rx_window = getattr(session, "rx_cts_window", 0) or getattr(self, "RX_CTS_WINDOW", 0)
 1037|         if rx_window > 0:
 1038|             grant = min(grant, rx_window)
 1039|         grant = min(grant, MAX_CTS_PACKET_COUNT)
 1040| 
 1041|         cts_data = bytearray(8)
 1042|         cts_data[0] = TP_CTRL_CTS
 1043|         cts_data[1] = max(1, grant)  # Number of packets allowed
 1044|         cts_data[2] = session.expected_sequence  # Next sequence number expected (1)
 1045|         cts_data[3] = 0xFF
 1046|         cts_data[4] = 0xFF
 1047|         cts_data[5:8] = session.target_pgn.to_bytes(3, byteorder="little")
 1048| 
 1049|         can_id = 0x1CEC0000 | (session.source_address << 8) | (self.my_address & 0xFF)
 1050|         return CanFrame.create(
 1051|             channel_id=session.channel_id,
 1052|             arbitration_id=can_id,
 1053|             data=bytes(cts_data),
 1054|             is_extended=True,
 1055|             direction="tx",
 1056|         )

 1132|     def start_tp_bam_paced(
 1133|         self,
 1134|         pgn: int,
 1135|         data: bytes,
 1136|         channel_id: str | None = None,
 1137|         interval_s: float | None = None,
 1138|     ) -> list[tuple[CanFrame, float]]:
 1139|         """BAM with SAE J1939-21 mandated 50-200 ms TP.DT pacing metadata.
 1140| 
 1141|         REVIEW 1-H5 (HIGH): BAM is a broadcast with no flow control; the
 1142|         standard mandates 50-200 ms between TP.DT packets so slow or

 1144|         undetectable — the receiver drops the whole transfer on T1

 1146|         earliest_tx_time_s) pairs on time.monotonic(): the CM_BAM

 1262|     def advance_cmdt_transfer(
 1263|         self, target_address: int, pgn: int, channel_id: str | None = None
 1264|     ) -> tuple[list[CanFrame], CanFrame | None]:
 1265|         """Produce the next TP.DT batch for a pending CMDT sender session.
 1266| 
 1267|         Called when TP.CM_CTS arrives. Returns (dt_frames, timeout_abort):
 1268|         - dt_frames: the packets granted by the CTS window, or [] while
 1269|           waiting for the final EndOfMsgACK.
 1270|         - timeout_abort: TP.Conn_Abort(reason=Timeout) when T2/T3 expired;
 1271|           the session is released alongside.
 1272|         """
 1273|         ch = channel_id or self.channel_id
 1274|         key = (self.my_address, target_address, pgn, ch)
 1275|         now = self._get_now()
 1276| 
 1277|         with self._sessions_lock:
 1278|             session = self._tx_sessions.get(key)
 1279|             if session is None:
 1280|                 return [], None
 1281| 
 1282|             if session.state == "WAIT_CTS":
 1283|                 if (now - session.last_activity_time) > self.T2_TIMEOUT_SEC:
 1284|                     abort = self._create_tx_abort_frame(session, ABORT_REASON_TIMEOUT)
 1285|                     self._tx_sessions.pop(key, None)
 1286|                     return [], abort
 1287|                 return [], None
 1288| 
 1289|             if session.state == "WAIT_ACK":
 1290|                 if (now - session.last_activity_time) > self.T3_TIMEOUT_SEC:
 1291|                     abort = self._create_tx_abort_frame(session, ABORT_REASON_TIMEOUT)
 1292|                     self._tx_sessions.pop(key, None)
 1293|                     return [], abort
 1294|                 return [], None
 1295| 
 1296|             # state == GRANTED: emit the CTS-granted window
 1297|             return self._emit_dt_window(session, now), None

 1299|     def _emit_dt_window(self, session: CmdtSenderSession, now: float) -> list[CanFrame]:
 1300|         """Emit up to the CTS-granted packet count; transition session state.
 1301| 
 1302|         Caller must hold _sessions_lock.
 1303|         """
 1304|         window_end = min(session.next_sequence + session.cts_window - 1, session.total_packets)
 1305|         dt_frames: list[CanFrame] = []
 1306|         can_id_dt = 0x1CEB0000 | ((session.destination_address & 0xFF) << 8) | (self.my_address & 0xFF)
 1307|         while session.next_sequence <= window_end:
 1308|             seq = session.next_sequence
 1309|             chunk = session.data[(seq - 1) * 7 : seq * 7]
 1310|             dt_data = bytearray(8)
 1311|             dt_data[0] = seq
 1312|             dt_data[1 : 1 + len(chunk)] = chunk
 1313|             for i in range(1 + len(chunk), 8):
 1314|                 dt_data[i] = 0xFF
 1315|             dt_frames.append(
 1316|                 CanFrame.create(
 1317|                     channel_id=session.channel_id,
 1318|                     arbitration_id=can_id_dt,
 1319|                     data=bytes(dt_data),
 1320|                     is_extended=True,
 1321|                     direction="tx",
 1322|                 )
 1323|             )
 1324|             session.next_sequence = seq + 1
 1325|             session.last_activity_time = now
 1326| 
 1327|         if session.next_sequence > session.total_packets:
 1328|             session.state = "WAIT_ACK"
 1329|         else:
 1330|             session.state = "WAIT_CTS"
 1331|         session.last_activity_time = now
 1332|         return dt_frames

 1334|     def _handle_tx_cm(self, frame: CanFrame, sa: int, da: int) -> tuple[list[CanFrame], CanFrame | None]:
 1335|         """Process TP.CM frames addressed to OUR pending CMDT sender sessions."""
 1336|         ctrl_byte = frame.data[0]
 1337|         target_pgn = int.from_bytes(frame.data[5:8], byteorder="little")
 1338|         key = (self.my_address, sa, target_pgn, frame.channel_id)
 1339| 
 1340|         with self._sessions_lock:
 1341|             session = self._tx_sessions.get(key)
 1342|             if session is None:
 1343|                 # MEDIUM-7: a CTS with no matching sender session is
 1344|                 # unsolicited — a classic spoof/injection signature.
 1345|                 if ctrl_byte == TP_CTRL_CTS:
 1346|                     self.anomaly_metrics.rx_cts += 1
 1347|                     self.anomaly_metrics.rx_unsolicited_cts += 1
 1348|                     logger.warning(
 1349|                         "J1939 unsolicited TP.CM_CTS (no matching sender session)",
 1350|                         extra={"sa": sa, "target_pgn": hex(target_pgn)},
 1351|                     )
 1352|                 return [], None
 1353| 
 1354|             if ctrl_byte == TP_CTRL_CTS:
 1355|                 self.anomaly_metrics.rx_cts += 1
 1356|                 packet_count = frame.data[1]
 1357|                 next_seq = frame.data[2]
 1358|                 # P2-6: SAE J1939-21 semantics — CTS with zero packets means
 1359|                 # "connection held, receiver not ready" (that is exactly what
 1360|                 # T4 exists for), and a lower next_seq is a retransmit
 1361|                 # request. Both are legal; only truly impossible sequences
 1362|                 # (0 or > total_packets) abort.
 1363|                 if next_seq == 0 or next_seq > session.total_packets or next_seq > session.next_sequence:
 1364|                     abort = self._create_tx_abort_frame(session, ABORT_REASON_UNEXPECTED_CONTROL)
 1365|                     self._tx_sessions.pop(key, None)
 1366|                     return [], abort
 1367|                 # REVIEW hardening: CTS grant cap — an inflated packet_count
 1368|                 # (up to 255) would burst the whole transfer in one window
 1369|                 # and overflow the receiver buffer. Grants above the cap
 1370|                 # abort the session (fail-closed, abort path preserved).
 1371|                 if packet_count > MAX_CTS_PACKET_COUNT:
 1372|                     logger.warning(
 1373|                         "J1939 CTS packet_count exceeds cap — aborting session",
 1374|                         extra={"packet_count": packet_count, "cap": MAX_CTS_PACKET_COUNT},
 1375|                     )
 1376|                     abort = self._create_tx_abort_frame(session, ABORT_REASON_UNEXPECTED_CONTROL)
 1377|                     self._tx_sessions.pop(key, None)
 1378|                     return [], abort
 1379|                 if packet_count == 0:
 1380|                     # HOLD: keep the session alive, refresh activity so T4
 1381|                     # governs the wait, and emit nothing until a real CTS.
 1382|                     session.state = "WAIT_CTS"
 1383|                     session.last_activity_time = self._get_now()
 1384|                     return [], None
 1385|                 if next_seq < session.next_sequence:
 1386|                     # Retransmit window: rewind to the requested sequence.
 1387|                     logger.info(
 1388|                         "J1939 CMDT retransmit requested by receiver CTS",
 1389|                         extra={"from_seq": session.next_sequence, "to_seq": next_seq},
 1390|                     )
 1391|                 session.cts_window = packet_count
 1392|                 session.next_sequence = next_seq
 1393|                 session.state = "GRANTED"
 1394|                 session.last_activity_time = self._get_now()
 1395|                 return self._emit_dt_window(session, self._get_now()), None
 1396| 
 1397|             if ctrl_byte == TP_CTRL_ACK:
 1398|                 # EndOfMsgACK: verify session state, bytes, packets, and PGN before closing
 1399|                 ack_bytes = int.from_bytes(frame.data[1:3], byteorder="little")
 1400|                 ack_packets = frame.data[3]
 1401|                 ack_pgn = int.from_bytes(frame.data[5:8], byteorder="little")
 1402| 
 1403|                 if (
 1404|                     session.state != "WAIT_ACK"
 1405|                     or ack_bytes != session.total_bytes
 1406|                     or ack_packets != session.total_packets
 1407|                     or ack_pgn != session.target_pgn
 1408|                 ):
 1409|                     logger.warning(
 1410|                         "Malformed or out-of-order TP.CM_ACK received",
 1411|                         extra={
 1412|                             "state": session.state,
 1413|                             "expected_bytes": session.total_bytes,
 1414|                             "actual_bytes": ack_bytes,
 1415|                             "expected_packets": session.total_packets,
 1416|                             "actual_packets": ack_packets,
 1417|                             "target_pgn": hex(session.target_pgn),
 1418|                             "ack_pgn": hex(ack_pgn),
 1419|                         },
 1420|                     )
 1421|                     abort = self._create_tx_abort_frame(session, ABORT_REASON_UNEXPECTED_CONTROL)
 1422|                     self._tx_sessions.pop(key, None)
 1423|                     return [], abort
 1424| 
 1425|                 # Valid EndOfMsgACK: transfer complete
 1426|                 self._tx_sessions.pop(key, None)
 1427|                 return [], None
 1428| 
 1429|             if ctrl_byte == TP_CTRL_ABORT:
 1430|                 logger.warning(
 1431|                     "Received TP.Conn_Abort for our CMDT transfer",
 1432|                     extra={"sa": sa, "target_pgn": hex(target_pgn), "reason": frame.data[1]},
 1433|                 )
 1434|                 self._tx_sessions.pop(key, None)

 1436|

  463|             )
  464|             ff_padded = pad_payload(ff_raw, 8, pad_byte=self.pad_byte if self.pad_byte is not None else 0xCC)
  465|             frames_classic.append(
  466|                 CanFrame(
  467|                     channel_id=self.channel_id,
  468|                     arbitration_id=self.tx_id,
  469|                     dlc=8,
  470|                     data=ff_padded,
  471|                     is_extended=self.tx_id > 0x7FF,
  472|                     is_fd=False,
  473|                     direction="tx",
  474|                 )
  475|             )
  476|             bytes_sent = ff_capacity
  477|         else:
  478|             # Extended 32-bit First Frame
  479|             # M-32 (P2-25): a classic 32-bit FF is only feasible when the
  480|             # PCI (2) + length (4) + addressing (addr) + data (2) fits the
  481|             # 8-byte payload — i.e. only NORMAL addressing (addr == 0). In
  482|             # EXTENDED/MIXED modes the padded classic frame would exceed 8
  483|             # bytes and pad_payload raised a bare ValueError mid-segmentation;
  484|             # ISO 15765-2 §9.2.3 requires CAN FD for such transfers. Fail
  485|             # closed with the structured protocol error instead.
  486|             if addr > 0:
  487|                 raise IsoTpError(
  488|                     "Classic-frame ISO-TP cannot carry >4095-byte payloads in "
  489|                     "EXTENDED/MIXED addressing — use CAN FD for this transfer"
  490|                 )
  491|             ff_raw = self._wrap_npci(bytes([0x10, 0x00]) + data_len.to_bytes(4, byteorder="big") + data[:2])
  492|             ff_padded = pad_payload(ff_raw, 8, pad_byte=self.pad_byte if self.pad_byte is not None else 0xCC)
  493|             frames_classic.append(
  494|                 CanFrame(
  495|                     channel_id=self.channel_id,
  496|                     arbitration_id=self.tx_id,
  497|                     dlc=8,
  498|                     data=ff_padded,
  499|                     is_extended=self.tx_id > 0x7FF,
  500|                     is_fd=False,
  501|                     direction="tx",
  502|                 )
  503|             )
  504|             # addr == 0 here, so the FF payload carries exactly 2 data bytes.
  505|             bytes_sent = 2
  506| 
  507|         seq_num = 1
  508|         while bytes_sent < data_len:
  509|             chunk = data[bytes_sent : bytes_sent + cf_capacity]
  510|             cf_raw = self._wrap_npci(bytes([(PCI_CONSECUTIVE_FRAME << 4) | (seq_num & 0x0F)]) + chunk)
  511|             cf_padded = pad_payload(cf_raw, 8, pad_byte=self.pad_byte if self.pad_byte is not None else 0xCC)
  512|             frames_classic.append(
  513|                 CanFrame(
  514|                     channel_id=self.channel_id,
  515|                     arbitration_id=self.tx_id,
  516|                     dlc=8,
  517|                     data=cf_padded,
  518|                     is_extended=self.tx_id > 0x7FF,
  519|                     is_fd=False,
  520|                     direction="tx",
  521|                 )
  522|             )
  523|             bytes_sent += len(chunk)
  524|             seq_num = (seq_num + 1) & 0x0F
  525| 
  526|         return frames_classic
  527| 
  528|     def handle_rx_frame(self, frame: CanFrame) -> tuple[bytes | None, CanFrame | None]:
  529|         """Process incoming CAN frame for ISO-TP reassembly (Thread-safe, B-01/CRITICAL-3).
  530| 
  531|         Returns: (CompletedPayload, ResponseFrame)
  532| 
  533|         ISO 15765-2 §9.2 addressing: in EXTENDED/MIXED modes byte 0 is the
  534|         N_TA/N_AE address byte and the N_PCI starts at byte 1; the address
  535|         byte is verified (mismatched frames are dropped) and stripped before
  536|         reassembly. In NORMAL mode the N_PCI is at byte 0 as before.
  537|         """
  538|         if frame.arbitration_id != self.rx_id or len(frame.data) < 2:
  539|             return None, None
  540| 
  541|         if self.addressing_mode != AddressingMode.NORMAL:
  542|             # Verify then strip the addressing byte (N_TA / N_AE)
  543|             if frame.data[0] != self.address_byte:
  544|                 logger.debug(
  545|                     "ISO-TP frame address byte mismatch — dropping",
  546|                     extra={"expected": self.address_byte, "got": frame.data[0]},
  547|                 )
  548|                 return None, None
  549|             data = frame.data[1:]
  550|             if len(data) < 1:
  551|                 return None, None
  552|         else:
  553|             data = frame.data
  554| 
  555|         pci_type = (data[0] >> 4) & 0x0F
  556|         now = time.monotonic()
  557|         key = (frame.arbitration_id, frame.channel_id)
  558| 
  559|         with self._lock:
  560|             # ------------------------------------------------------------------
  561|             # 1. Single Frame (SF)
  562|             # ------------------------------------------------------------------
  563|             if pci_type == PCI_SINGLE_FRAME:
  564|                 if (data[0] & 0x0F) == 0:
  565|                     # If frame is not CAN-FD, SF_DL == 0 is malformed Classic CAN frame
  566|                     if not frame.is_fd:
  567|                         return None, None
  568| 
  569|                     # P2-5: the escape form requires CAN_DL > 8 — an 8-byte FD
  570|                     # frame carrying 00 xx is malformed and must be dropped.
  571|                     if len(frame.data) <= 8:
  572|                         return None, None
  573| 
  574|                     # CAN-FD Extended Single Frame (SF_DL 8..62)
  575|                     if len(data) < 2:
  576|                         return None, None
  577|                     sf_len = data[1]
  578|                     if sf_len < 8 or sf_len > 62:
  579|                         return None, None
  580|                     if sf_len <= (len(data) - 2):
  581|                         self._sessions.pop(key, None)
  582|                         return bytes(data[2 : 2 + sf_len]), None
  583|                 else:
  584|                     # Classic CAN Single Frame (SF_DL 1..7)
  585|                     sf_len = data[0] & 0x0F
  586|                     if 1 <= sf_len <= (len(data) - 1):
  587|                         self._sessions.pop(key, None)
  588|                         return bytes(data[1 : 1 + sf_len]), None
  589|                 return None, None
  590| 
  591|             # ------------------------------------------------------------------
  592|             # 2. First Frame (FF)
  593|             # ------------------------------------------------------------------
  594|             if pci_type == PCI_FIRST_FRAME:
  595|                 # REVIEW (short-frame guard): after the address strip the
  596|                 # N_PCI view can be as short as one byte; every branch below
  597|                 # indexes data[1] (and data[2:6] for the 32-bit form), so
  598|                 # enforce the PCI-specific minimum length up front instead
  599|                 # of crashing with IndexError.
  600|                 if len(data) < 2:
  601|                     return None, None
  602|                 if len(data) >= 6 and data[0] == 0x10 and data[1] == 0x00:
  603|                     # Extended 32-bit First Frame
  604|                     total_len = int.from_bytes(data[2:6], byteorder="big")
  605|                     if total_len <= 4095:
  606|                         return None, None
  607|                     header_len = 6
  608|                 else:
  609|                     # Standard 12-bit First Frame (ISO 15765-2 requires total_len >= 8)
  610|                     total_len = ((data[0] & 0x0F) << 8) | data[1]
  611|                     if total_len < 8:
  612|                         return None, None
  613|                     header_len = 2
  614| 
  615|                 # P1: Check buffer size cap (ISO 15765-2 FS_OVERFLOW)
  616|                 if total_len > self.max_buffer_size:
  617|                     logger.warning(
  618|                         "ISO-TP First Frame length exceeds max buffer size. Emitting FlowControl OVERFLOW.",
  619|                         extra={"total_len": total_len, "max_buffer_size": self.max_buffer_size},
  620|                     )
  621|                     self._sessions.pop(key, None)
  622|                     # REVIEW (FC frame budget): the N_TA/N_AE prefix consumes
  623|                     # one of the 8 classic payload bytes — an 8-byte fc_data
  624|                     # plus the prefix made a 9-byte frame that CanFrame
  625|                     # rejects. FC needs only 3 real bytes; pad to 8 - addr.
  626|                     fc_data = bytearray(8 - self._pci_offset())
  627|                     fc_data[0] = (PCI_FLOW_CONTROL << 4) | FS_OVERFLOW
  628|                     for i in range(1, len(fc_data)):
  629|                         fc_data[i] = 0xCC
  630| 
  631|                     fc_frame = CanFrame.create(
  632|                         channel_id=frame.channel_id,
  633|                         arbitration_id=self.tx_id,
  634|                         # REVIEW (FC addressing): OVERFLOW FCs carry the
  635|                         # N_TA/N_AE prefix like every other FC (see CTS site).
  636|                         data=self._wrap_npci(bytes(fc_data)),
  637|                         # M-6 (P1-12): the FC's addressing must derive from the
  638|                         # TRANSMIT id, not the received frame — a mixed setup
  639|                         # (11-bit tx_id, 29-bit rx_id) otherwise emitted the FC
  640|                         # as an extended frame carrying an 11-bit request ID
  641|                         # the ECU never receives.
  642|                         is_extended=self.tx_id > 0x7FF,
  643|                         is_fd=frame.is_fd,
  644|                         direction="tx",
  645|                     )
  646|                     return None, fc_frame
  647| 
  648|                 first_chunk = data[header_len:]
  649|                 if len(first_chunk) > total_len:
  650|                     first_chunk = first_chunk[:total_len]
  651| 
  652|                 if key in self._sessions:
  653|                     logger.warning(
  654|                         "New ISO-TP First Frame aborts previous incomplete session",
  655|                         extra={"rx_id": hex(frame.arbitration_id), "channel": frame.channel_id},
  656|                     )
  657| 
  658|                 # REVIEW hardening: bounded session table + stale-FF reap.
  659|                 # Without a cap a bus attacker mints sessions with distinct
  660|                 # channel_ids until RAM is gone; without reap a stale FF
  661|                 # pins its slot until a CF arrives.
  662|                 self._reap_stale_rx_sessions(now)
  663|                 if key not in self._sessions and len(self._sessions) >= MAX_RX_SESSIONS:
  664|                     # Evict the stalest session (fail-closed: never grow
  665|                     # unbounded, never drop the live key being replaced).
  666|                     oldest_key = min(self._sessions.keys(), key=lambda k: self._sessions[k].last_activity_time)
  667|                     logger.warning(
  668|                         "ISO-TP session table full — evicting stalest session",
  669|                         extra={"cap": MAX_RX_SESSIONS},
  670|                     )
  671|                     self._sessions.pop(oldest_key, None)
  672| 
  673|                 new_session = IsoTpRxSession(
  674|                     rx_id=frame.arbitration_id,
  675|                     total_bytes=total_len,
  676|                     expected_sequence_number=1,
  677|                     received_bytes=bytearray(first_chunk),
  678|                     last_activity_time=now,
  679|                     channel_id=frame.channel_id,
  680|                     is_fd=frame.is_fd,
  681|                 )
  682|                 self._sessions[key] = new_session
  683| 
  684|                 # If First Frame already satisfied full payload length (e.g. 62B FD)
  685|                 if len(new_session.received_bytes) >= total_len:
  686|                     completed = bytes(new_session.received_bytes[:total_len])
  687|                     self._sessions.pop(key, None)
  688|                     return completed, None
  689| 
  690|                 # Generate Flow Control (CTS with configured BS/STmin)
  691|                 # REVIEW (FC frame budget): see OVERFLOW site — the address
  692|                 # prefix must fit inside the 8-byte classic payload.
  693|                 fc_data = bytearray(8 - self._pci_offset())
  694|                 fc_data[0] = (PCI_FLOW_CONTROL << 4) | FS_CTS
  695|                 fc_data[1] = self.rx_block_size & 0xFF  # Block size (0 = send all)
  696|                 fc_data[2] = self.rx_st_min & 0xFF  # STmin (raw ISO 15765-2 byte)
  697|                 for i in range(3, len(fc_data)):
  698|                     fc_data[i] = 0xCC
  699| 
  700|                 fc_frame = CanFrame.create(
  701|                     channel_id=frame.channel_id,
  702|                     arbitration_id=self.tx_id,
  703|                     # REVIEW (FC addressing): Flow Control frames in
  704|                     # EXTENDED/MIXED addressing MUST carry the N_TA/N_AE
  705|                     # prefix byte — the ECU drops a bare `30 00 00..` FC and
  706|                     # the sender dies in N_Bs timeout.
  707|                     data=self._wrap_npci(bytes(fc_data)),
  708|                     # M-6 (P1-12): addressing derives from tx_id (see OVERFLOW site).
  709|                     is_extended=self.tx_id > 0x7FF,
  710|                     is_fd=frame.is_fd,
  711|                     direction="tx",
  712|                 )
  713|                 return None, fc_frame
  714| 
  715|             # ------------------------------------------------------------------
  716|             # 3. Consecutive Frame (CF)
  717|             # ------------------------------------------------------------------
  718|             if pci_type == PCI_CONSECUTIVE_FRAME:
  719|                 session = self._sessions.get(key)
  720|                 if session is None:
  721|                     return None, None
  722| 
  723|                 # F-20: a CF arriving on a different channel must not corrupt an
  724|                 # in-flight session opened on another bus.
  725|                 if session.channel_id != frame.channel_id:
  726|                     logger.warning(
  727|                         "ISO-TP CF channel mismatch — ignoring frame",
  728|                         extra={"session_channel": session.channel_id, "frame_channel": frame.channel_id},
  729|                     )
  730|                     return None, None
  731| 
  732|                 # Check N_Cr timeout
  733|                 if (now - session.last_activity_time) > self.TIMEOUT_SEC:
  734|                     logger.warning("ISO-TP Consecutive Frame timeout", extra={"rx_id": hex(self.rx_id)})
  735|                     self._sessions.pop(key, None)
  736|                     return None, None
  737| 
  738|                 seq_num = data[0] & 0x0F
  739|                 if seq_num != session.expected_sequence_number:
  740|                     logger.warning(
  741|                         "ISO-TP Sequence mismatch",
  742|                         extra={"expected": session.expected_sequence_number, "got": seq_num},
  743|                     )
  744|                     self._sessions.pop(key, None)
  745|                     return None, None
  746| 
  747|                 needed = session.total_bytes - len(session.received_bytes)
  748|                 available_payload = len(data) - 1
  749|                 chunk_len = min(needed, available_payload)
  750|                 chunk = data[1 : 1 + chunk_len]
  751| 
  752|                 session.received_bytes.extend(chunk)
  753|                 session.expected_sequence_number = (session.expected_sequence_number + 1) & 0x0F
  754|                 session.last_activity_time = now
  755| 
  756|                 if len(session.received_bytes) >= session.total_bytes:
  757|                     completed = bytes(session.received_bytes[: session.total_bytes])
  758|                     self._sessions.pop(key, None)
  759|                     return completed, None
  760| 
  761|                 # BS windowing: after rx_block_size consecutive frames the sender
  762|                 # needs a fresh FC (CTS) before it may continue transmitting.
  763|                 if self.rx_block_size > 0:
  764|                     session.block_count += 1
  765|                     if session.block_count >= self.rx_block_size:
  766|                         session.block_count = 0
  767|                         # REVIEW (FC frame budget): see OVERFLOW site.
  768|                         fc_data = bytearray(8 - self._pci_offset())
  769|                         fc_data[0] = (PCI_FLOW_CONTROL << 4) | FS_CTS
  770|                         fc_data[1] = self.rx_block_size & 0xFF
  771|                         fc_data[2] = self.rx_st_min & 0xFF
  772|                         for i in range(3, len(fc_data)):
  773|                             fc_data[i] = 0xCC
  774| 
  775|                         fc_frame = CanFrame.create(
  776|                             channel_id=frame.channel_id,
  777|                             arbitration_id=self.tx_id,
  778|                             # REVIEW (FC addressing): windowed CTS FCs carry the
  779|                             # N_TA/N_AE prefix like every other FC (see CTS site).
  780|                             data=self._wrap_npci(bytes(fc_data)),
  781|                             # M-6 (P1-12): addressing derives from tx_id (see OVERFLOW site).
  782|                             is_extended=self.tx_id > 0x7FF,
  783|                             is_fd=frame.is_fd,
  784|                             direction="tx",
  785|                         )
  786|                         return None, fc_frame
  787| 
  788|                 return None, None
  789| 
  790|         return None, None

  831|     def _build_frame(self, data: bytes, is_fd: bool) -> CanFrame:
  832|         """Construct normalized and padded CAN frame."""
  833|         # channel_id=None (any-channel FC matching) still needs a concrete
  834|         # id on outgoing frames — CanFrame's pattern validation rejects "".
  835|         out_channel = self.channel_id or "uds_ch0"
  836|         if is_fd:
  837|             dlc = length_to_dlc(len(data))
  838|             padded = pad_payload(data, dlc, pad_byte=self.pad_byte if self.pad_byte is not None else 0xCC)
  839|             return CanFrame(
  840|                 channel_id=out_channel,
  841|                 arbitration_id=self.tx_id,
  842|                 dlc=dlc,
  843|                 data=padded,
  844|                 is_extended=self.tx_id > 0x7FF,
  845|                 is_fd=True,
  846|                 direction="tx",
  847|             )
  848|         else:
  849|             padded = pad_payload(data, 8, pad_byte=self.pad_byte if self.pad_byte is not None else 0xCC)
  850|             return CanFrame(
  851|                 channel_id=out_channel,
  852|                 arbitration_id=self.tx_id,
  853|                 dlc=8,
  854|                 data=padded,
  855|                 is_extended=self.tx_id > 0x7FF,
  856|                 is_fd=False,
  857|                 direction="tx",
  858|             )

  860|     async def _send_with_n_as(self, frame: CanFrame) -> None:
  861|         """Transmit one frame enforcing the N_As timeout.
  862| 
  863|         N_As (ISO 15765-2 §4.6.1) bounds the time for the network layer to
  864|         complete a single frame transmission after the request. A TxPort
  865|         that blocks longer than n_as_timeout_s is treated as a timeout.
  866| 
  867|         P2-9: cancelling a `run_in_executor` future does NOT stop the
  868|         underlying thread — the frame can still hit the bus AFTER the
  869|         caller has given up and retried, interleaving with the new
  870|         attempt. After a timeout we therefore DRAIN the in-flight task to
  871|         completion before raising, so a late frame either lands before the
  872|         error surfaces (and the caller knows the exchange is dead) or the
  873|         send itself failed. The bus ordering stays consistent with the
  874|         session state either way.
  875|         """
  876|         send_task: asyncio.Future[None] = asyncio.ensure_future(self.tx_port.send(frame))
  877|         start = self.clock.now_monotonic()
  878|         try:
  879|             await asyncio.wait_for(asyncio.shield(send_task), timeout=self.n_as_timeout_s)
  880|             return
  881|         except asyncio.TimeoutError:
  882|             # P2-9: wait (bounded) for the executor thread to finish so no
  883|             # frame slips onto the wire after we raise. Give the driver a
  884|             # short grace period beyond N_As before abandoning the drain.
  885|             try:
  886|                 await asyncio.wait_for(asyncio.shield(send_task), timeout=self.n_as_timeout_s)
  887|                 # The frame DID go out — but too late for the protocol
  888|                 # timing; the session is invalid, surface the timeout.
  889|                 raise IsoTpTimeoutError(
  890|                     f"N_As timeout ({self.n_as_timeout_s * 1000:.0f}ms) transmitting ISO-TP frame (late completion)",
  891|                     timeout_type="N_As",
  892|                     elapsed_ms=(self.clock.now_monotonic() - start) * 1000.0,
  893|                     limit_ms=self.n_as_timeout_s * 1000.0,
  894|                 )
  895|             except asyncio.TimeoutError:
  896|                 # Still stuck after the grace drain: cancel and let the
  897|                 # executor thread finish on its own — nothing we can do,
  898|                 # but we never overlap a retry with this attempt because
  899|                 # the caller must treat this session as dead.
  900|                 send_task.cancel()
  901|                 raise IsoTpTimeoutError(
  902|                     f"N_As timeout ({self.n_as_timeout_s * 1000:.0f}ms) transmitting ISO-TP frame",
  903|                     timeout_type="N_As",
  904|                     elapsed_ms=(self.clock.now_monotonic() - start) * 1000.0,
  905|                     limit_ms=self.n_as_timeout_s * 1000.0,
  906|                 ) from None

  908|     async def _apply_st_min(self, st_min_byte: int) -> None:
  909|         """Execute STmin pacing delay (all variants yield to the event loop).
  910| 
  911|         4.3 (REVIEW): the old 100-900 us range used a busy spin-wait
  912|         (`while ... pass`) INSIDE the async sender — that froze every other
  913|         coroutine on the thread (timers, UI pushes, FC readers) during high
  914|         frequency transfers. Yielding costs at most one event-loop tick of
  915|         scheduling latency (~1 ms), which is within STmin tolerance.
  916| 
  917|         B10 (REVIEW): the residual spin-wait for the sub-millisecond
  918|         0xF1..0xF9 range is replaced with a chunked wait — the loop yields
  919|         to the event loop (`await asyncio.sleep(0)`) every ~200 us so other
  920|         coroutines keep running, while retaining high-resolution pacing that
  921|         Windows' 15.6 ms sleep granularity cannot provide.
  922|         """
  923|         if st_min_byte == 0x00:
  924|             return
  925|         elif 0x01 <= st_min_byte <= 0x7F:
  926|             delay_s = st_min_byte / 1000.0
  927|             await asyncio.sleep(delay_s)
  928|         elif 0xF1 <= st_min_byte <= 0xF9:
  929|             # 100..900 us: high-resolution pacing with periodic event-loop
  930|             # yields; a full busy spin would starve the loop for the whole
  931|             # transfer and trip N_Bs/N_Cr timers. The yield interval is
  932|             # below the smallest STmin (100 us) so even the fastest pacing
  933|             # grants the loop a tick between waits.
  934|             delay_us = (st_min_byte - 0xF0) * 100
  935|             yield_every_ns = 50_000
  936|             target_ns = time.perf_counter_ns() + (delay_us * 1000)
  937|             while True:
  938|                 now_ns = time.perf_counter_ns()
  939|                 if now_ns >= target_ns:
  940|                     break
  941|                 if target_ns - now_ns > yield_every_ns:
  942|                     await asyncio.sleep(0)
  943|         else:
  944|             # Reserved range clamped to 127 ms
  945|             await asyncio.sleep(0.127)
  946| 
  947|     async def _await_flow_control(self) -> CanFrame:
  948|         """Await incoming Flow Control frame enforcing N_Bs timeout."""
  949|         start_time = self.clock.now_monotonic()

  188|     def _call_transfer_data(self, block_sequence: int, data: bytes, user_confirmed: bool = False) -> Any:

  191|         P1-1 (REVIEW M-8/H-4): the operator's dual confirmation is granted

  208|                 is_critical_command=True,

  230|     @staticmethod
  231|     def _parse_max_block_length(download_response: Any) -> int:
  232|         """Parse maxNumberOfBlockLength from a positive 0x34 response (P1-6).
  233| 
  234|         Response layout per ISO 14229-1 §14.2:
  235|             lengthFormatIdentifier (1B) | maxNumberOfBlockLength (n bytes)
  236| 
  237|         The high nibble encodes the width (1..8, 0xF = variable) and the
  238|         low nibble the address unit size. An unparsable/absent value fails
  239|         closed — the caller must never flash with an unbounded block size.
  240|         """
  241|         data = getattr(download_response, "data", None)
  242|         if not data or len(data) < 1:
  243|             raise ProtocolError(
  244|                 "0x34 yanıtı maxNumberOfBlockLength içermiyor — sınırsız blok boyutu ile flash reddedildi (fail-closed)"
  245|             )
  246|         length_format = data[0]
  247|         num_bytes = (length_format >> 4) & 0x0F
  248|         if not (1 <= num_bytes <= 8):
  249|             raise ProtocolError(
  250|                 f"0x34 yanıtı geçersiz lengthFormatIdentifier (0x{length_format:02X}) — flash reddedildi (fail-closed)"
  251|             )
  252|         raw = bytes(data[1 : 1 + num_bytes])
  253|         if len(raw) < num_bytes:
  254|             raise ProtocolError("0x34 yanıtı maxNumberOfBlockLength alanı kesik — flash reddedildi (fail-closed)")
  255|         value = int.from_bytes(raw, byteorder="big")
  256|         if value <= 0:
  257|             raise ProtocolError("0x34 yanıtı maxNumberOfBlockLength=0 — flash reddedildi (fail-closed)")
  258|         # REVIEW hardening: ECU-controlled bound — clamp to the tool-side
  259|         # ceiling so a malicious/broken ECU cannot force OOM-size blocks.
  260|         if value > FlashingConfig.MAX_BLOCK_LENGTH_CAP:
  261|             logger.warning(
  262|                 "0x34 maxNumberOfBlockLength tavana çekildi (ECU değeri sınırlı)",
  263|                 extra={
  264|                     "ecu_value": value,
  265|                     "cap": FlashingConfig.MAX_BLOCK_LENGTH_CAP,
  266|                 },
  267|             )
  268|             return FlashingConfig.MAX_BLOCK_LENGTH_CAP
  269|         return value

  271|     def _assert_gateway_preconditions(self) -> None:
  272|         """P1-8: verify every TX precondition before touching the ECU.
  273| 
  274|         Once the first session-control frame leaves, a mid-sequence gateway
  275|         rejection leaves the ECU in a programming session with the recovery
  276|         reset itself blocked (ESTOP_ACTIVE). Checking supervisor state,
  277|         watchdog lease, and speed interlock up front turns those failures
  278|         into step-1 SafetyErrors — before the ECU is disturbed.
  279|         """
  280|         estop = self.gateway.estop
  281|         if estop.is_engaged:
  282|             raise SafetyError("E-Stop aktifken flashing yapılamaz!")
  283| 
  284|         supervisor = getattr(self.gateway, "supervisor", None)
  285|         if supervisor is not None and not getattr(supervisor, "is_tx_permitted", True):
  286|             raise SafetyError(
  287|                 "Güvenlik süpervizörü TX izni vermiyor (FAULT/ARMED olmayan durum) — flashing reddedildi."
  288|             )
  289| 
  290|         watchdog = getattr(self.gateway, "watchdog", None)
  291|         if watchdog is not None and not getattr(watchdog, "is_lease_valid", True):
  292|             raise SafetyError(
  293|                 "TX watchdog kira süresi geçersiz — flashing reddedildi (lease yenileyin)."
  294|             )
  295| 
  296|         # Speed interlock: check through public gateway API when available
  297|         if hasattr(self.gateway, "is_speed_fresh_and_safe"):
  298|             if not self.gateway.is_speed_fresh_and_safe():
  299|                 raise SafetyError("Hız telemetrisi yok/taze değil veya araç hareket halinde — flashing reddedildi.")
  300|         else:
  301|             # Fallback for mock gateways in unit tests
  302|             last_update_ns = getattr(self.gateway, "_last_speed_update_ns", None)
  303|             if not isinstance(last_update_ns, int) or last_update_ns == 0:
  304|                 raise SafetyError("Hız telemetrisi yok/taze değil — hareketli araçta flashing reddedildi.")
  305|             speed = getattr(self.gateway, "_current_vehicle_speed_kmh", 0.0)
  306|             threshold = getattr(self.gateway, "SPEED_NOISE_THRESHOLD_KMH", 0.5)
  307|             if not isinstance(speed, (int, float)) or isinstance(speed, bool):
  308|                 raise SafetyError("Hız telemetrisi geçersiz (non-numeric) — flashing reddedildi.")
  309|             if math.isnan(float(speed)) or math.isinf(float(speed)):
  310|                 raise SafetyError("Hız telemetrisi geçersiz (NaN/Inf) — flashing reddedildi.")
  311|             if not isinstance(threshold, (int, float)) or isinstance(threshold, bool):
  312|                 threshold = 0.5
  313|             if float(speed) > float(threshold):
  314|                 raise SafetyError(
  315|                     f"Araç hareket halinde ({float(speed):.1f} km/s > {float(threshold)} km/s) — flashing reddedildi."
  316|                 )

  328|     def _emit_progress(
  329|         self,
  330|         step: FlashingStep,
  331|         step_idx: int,
  332|         bytes_sent: int,
  333|         total_bytes: int,
  334|         start_time: float,
  335|         crc_str: str,
  336|     ) -> None:
  337|         self.current_step = step
  338|         now = time.monotonic()
  339|         elapsed = max(0.001, now - start_time)
  340|         pct = (bytes_sent / total_bytes * 100.0) if total_bytes > 0 else 0.0
  341|         speed_kbps = (bytes_sent / 1024.0) / elapsed
  342| 
  343|         progress = FlashingProgress(
  344|             current_step=step,
  345|             step_index=step_idx,
  346|             total_steps=self.TOTAL_STEPS,
  347|             bytes_transferred=bytes_sent,
  348|             total_bytes=total_bytes,
  349|             percent=round(pct, 1),
  350|             transfer_speed_kbps=round(speed_kbps, 2),
  351|             elapsed_time_s=round(elapsed, 2),
  352|             crc32_checksum=crc_str,
  353|         )
  354| 
  355|         if self.on_progress:
  356|             self.on_progress(progress)

  358|     # REVIEW.md 3.3: S3 tester-present cadence. ISO 14229 S3 is 5 s; send at
  359|     # 2 s so one lost/deferred TesterPresent never drops the session during
  360|     # long erase/key-computation gaps.
  361|     TESTER_PRESENT_INTERVAL_S: ClassVar[float] = 2.0
  362| 
  363|     def _tester_present_loop(self, stop_event: threading.Event, config: FlashingConfig) -> None:
  364|         """Background keep-alive: TesterPresent 0x3E (suppress) while flashing.
  365| 
  366|         REVIEW.md 3.3: in extended/programming sessions the ECU runs the S3
  367|         timer (2-5 s). A 0x36 block whose erase cycle stalls the bus for a
  368|         few seconds otherwise returns the ECU to the default session, and
  369|         the NEXT TransferData gets rejected with NRC 0x7E — a half-written
  370|         flash. This loop keeps the session alive for the whole sequence.
  371|         """
  372|         while not stop_event.wait(self.TESTER_PRESENT_INTERVAL_S):
  373|             try:
  374|                 # suppressPosResponse bit (0x80): fire-and-forget, no reply
  375|                 # is expected and none is waited for.
  376|                 self.uds_client.tester_present(suppress_response=True)
  377|             except Exception as exc:  # noqa: BLE001 — keep-alive must never kill the flash
  378|                 logger.debug("TesterPresent keep-alive failed (retrying)", extra={"error": str(exc)})
  379| 
  380|     def execute_flash(self, config: FlashingConfig) -> bool:
  381|         """Execute full end-to-end ECU flashing cycle synchronously."""
  382|         self._is_cancelled = False
  383|         start_time = time.monotonic()
  384|         total_bytes = len(config.data)
  385|         if total_bytes == 0:
  386|             raise ValueError("Flashing payload data is empty")
  387| 
  388|         crc32_val = zlib.crc32(config.data) & 0xFFFFFFFF
  389|         crc_hex = f"0x{crc32_val:08X}"
  390|         # REVIEW hardening: SHA-256 alongside CRC32 (transport integrity vs
  391|         # cryptographic identity). CRC32 alone is forgeable; the SHA-256 is
  392|         # audit-logged for every flash.
  393|         sha256_hex = hashlib.sha256(bytes(config.data)).hexdigest()
  394|         self._log(
  395|             f"İmaj bütünlüğü: CRC32={crc_hex} SHA-256={sha256_hex} Boyut={total_bytes}",
  396|             "info",
  397|         )
  398| 
  399|         # REVIEW hardening (fail-log, not fail-mandatory yet): optional
  400|         # tool-side firmware signature + anti-rollback floor.
  401|         if config.firmware_signature is None:
  402|             self._log(
  403|                 "UYARI: firmware_signature yok — imza doğrulaması atlandı "
  404|                 "(fail-log; tam zorunluluk ileride).",
  405|                 "warning",
  406|             )
  407|         else:
  408|             sig_len = len(config.firmware_signature)
  409|             if sig_len == 0:
  410|                 raise SafetyError(
  411|                     "Boş firmware_signature ile flash reddedildi (fail-closed).",
  412|                     code="FLASH_SIGNATURE_INVALID",
  413|                 )
  414|             self._log(
  415|                 f"İmza alanı mevcut (uzunluk={sig_len}, min_version={config.min_version}).",
  416|                 "info",
  417|             )
  418| 
  419|         # REVIEW.md 3.3: start the S3 keep-alive before the first session
  420|         # control frame; stop it on completion, failure, or cancellation.
  421|         keepalive_stop = threading.Event()
  422|         keepalive_thread = threading.Thread(
  423|             target=self._tester_present_loop,
  424|             args=(keepalive_stop, config),
  425|             daemon=True,
  426|             name="uds_tester_present",
  427|         )
  428|         keepalive_thread.start()
  429| 
  430|         try:
  431|             return self._execute_flash_inner(config, start_time, crc32_val, crc_hex, sha256_hex)
  432|         finally:
  433|             keepalive_stop.set()
  434|             keepalive_thread.join(timeout=self.TESTER_PRESENT_INTERVAL_S * 2)
  435| 
  436|     def _execute_flash_inner(
  437|         self,
  438|         config: FlashingConfig,
  439|         start_time: float,
  440|         crc32_val: int,
  441|         crc_hex: str,
  442|         sha256_hex: str,
  443|     ) -> bool:
  444|         total_bytes = len(config.data)
  445|         self._log(
  446|             f"🚀 Flashing Başlatılıyor: Boyut={total_bytes} bayt, Hedef Adres=0x{config.memory_address:08X}, CRC32={crc_hex}",
  447|             "info",
  448|         )
  449| 
  450|         recovery_needed = False
  451|         try:
  452|             # 1. Safety Validation
  453|             self._emit_progress(FlashingStep.SAFETY_VALIDATION, 1, 0, total_bytes, start_time, crc_hex)
  454|             self._log("Adım 1/10: Hız kilidi ve güvenlik kontrolleri yapılıyor...", "info")
  455|             if not config.user_confirmed:
  456|                 raise SafetyError("Flashing işlemi operatörün açık çift onayını (Dual-Confirmation) gerektirir.")
  457| 
  458|             if self.gateway.estop.is_engaged:
  459|                 raise SafetyError("Acil Durdurma (E-Stop) aktifken flashing yapılamaz!")
  460| 
  461|             # P1-8: verify ALL gateway preconditions BEFORE a single byte
  462|             # reaches the ECU. Once the ECU leaves its default session a
  463|             # failed TX must recover it — cheaper to fail here, at step 1.
  464|             self._assert_gateway_preconditions()
  465| 
  466|             # 2. Extended Diagnostic Session
  467|             self._emit_progress(FlashingStep.EXTENDED_SESSION, 2, 0, total_bytes, start_time, crc_hex)
  468|             self._log("Adım 2/10: Genişletilmiş Diyagnostik Oturumu (0x10 0x03) açılıyor...", "info")
  469|             self._check_cancelled()
  470|             # REVIEW 3-CRITICAL: no TypeError fallback — a client signature
  471|             # without user_confirmed is a contract violation and must fail
  472|             # hard, never silently re-issue the request with the dual
  473|             # confirmation flag dropped (gateway Stage-5 bypass).
  474|             resp = self.uds_client.change_session(
  475|                 DiagnosticSessionType.EXTENDED_DIAGNOSTIC_SESSION, user_confirmed=True
  476|             )
  477|             if not resp.is_positive:
  478|                 raise ProtocolError(f"Genişletilmiş oturum açılamadı: {resp.nrc_description_tr} (NRC 0x{resp.nrc:02X})")
  479|             # From here on the ECU is out of its default session; a failure
  480|             # must attempt best-effort recovery before surfacing the error.
  481|             recovery_needed = True
  482| 
  483|             # Step 2b: Target Identity Verification (VIN / Serial DID match)
  484|             if config.expected_vin is not None:
  485|                 self._log(f"Hedef VIN doğrulaması yapılıyor (Beklenen: {config.expected_vin})...", "info")
  486|                 vin_resp = self.uds_client.read_did(0xF190)
  487|                 if not vin_resp.is_positive:
  488|                     raise ProtocolError(f"ECU VIN (0xF190) okunamadı: {vin_resp.nrc_description_tr}")
  489|                 raw_vin = vin_resp.data[2:] if len(vin_resp.data) >= 2 and vin_resp.data[:2] == b"\xf1\x90" else vin_resp.data
  490|                 actual_vin = raw_vin.decode("ascii", errors="replace").strip()
  491|                 if actual_vin != config.expected_vin:
  492|                     raise SafetyError(
  493|                         f"Araç VIN uyuşmazlığı! Beklenen: {config.expected_vin}, Okunan: {actual_vin} (Flashing fail-closed reddedildi)"
  494|                     )
  495|                 self._log(f"✅ VIN başarıyla doğrulandı: {actual_vin}", "info")
  496| 
  497|             if config.expected_serial is not None:
  498|                 self._log(f"Hedef Seri Numarası doğrulaması yapılıyor (Beklenen: {config.expected_serial})...", "info")
  499|                 ser_resp = self.uds_client.read_did(0xF18C)
  500|                 if not ser_resp.is_positive:
  501|                     raise ProtocolError(f"ECU Seri No (0xF18C) okunamadı: {ser_resp.nrc_description_tr}")
  502|                 raw_ser = ser_resp.data[2:] if len(ser_resp.data) >= 2 and ser_resp.data[:2] == b"\xf1\x8c" else ser_resp.data
  503|                 actual_serial = raw_ser.decode("ascii", errors="replace").strip()
  504|                 if actual_serial != config.expected_serial:
  505|                     raise SafetyError(
  506|                         f"ECU Seri No uyuşmazlığı! Beklenen: {config.expected_serial}, Okunan: {actual_serial} (Flashing fail-closed reddedildi)"
  507|                     )
  508|                 self._log(f"✅ ECU Seri Numarası başarıyla doğrulandı: {actual_serial}", "info")
  509| 
  510|             # 3. Programming Session (P1-5: 0x10 0x02 BEFORE 0x27 — the ECU
  511|             # re-locks security access on session transition; the normative

   93|         # REVIEW 1-M3 (MEDIUM): NRC 0x78 pending budget. ISO 14229-1 sets no
   94|         # count limit; the server may keep extending within its announced
   95|         # P2* window. A hard 30 s absolute cap killed healthy long routines
   96|         # (flash erase/checksum) mid-flight. Default: no absolute cap — each
   97|         # pending response re-arms the P2* window; an outer wall remains
   98|         # bounded by the per-request timeout_s.
   99|         self.max_pending_timeout_s = max_pending_timeout_s
  100|         self.transport = IsoTpTransport(tx_id=tx_id, rx_id=rx_id, channel_id=channel_id)
  101|         self._executor = concurrent.futures.ThreadPoolExecutor(max_workers=max_workers, thread_name_prefix="uds_client")
  102|         # M-07: UDS exchanges are stateful (session type, security seed/key
  103|         # ladder) — async and sync operations are serialized with an RLock so
  104|         # concurrent calls cannot corrupt another exchange's request/response.
  105|         self._operation_lock = threading.RLock()

  107|     def execute_async(
  108|         self,
  109|         fn: Callable[..., Any],
  110|         *args: Any,
  111|         callback: Callable[[Any], None] | None = None,
  112|         error_callback: Callable[[Exception], None] | None = None,
  113|         **kwargs: Any,
  114|     ) -> concurrent.futures.Future[Any]:
  115|         """Execute a diagnostic routine asynchronously in the background thread pool.
  116| 
  117|         M-7: the wrapped routine runs under the client-wide operation lock —
  118|         the UDS request/response dialogue (including any multi-frame FC
  119|         handshakes) is a single serialized conversation with the ECU.
  120|         """
  121| 
  122|         def _worker() -> Any:
  123|             try:
  124|                 with self._operation_lock:
  125|                     result = fn(*args, **kwargs)
  126|                 if callback is not None:
  127|                     try:
  128|                         callback(result)
  129|                     except Exception as cb_err:
  130|                         logger.error("Error in UdsClient async callback", extra={"error": str(cb_err)})
  131|                 return result
  132|             except Exception as exc:
  133|                 if error_callback is not None:
  134|                     try:
  135|                         error_callback(exc)
  136|                     except Exception as err_cb_err:
  137|                         logger.error("Error in UdsClient async error_callback", extra={"error": str(err_cb_err)})
  138|                 raise
  139| 
  140|         return self._executor.submit(_worker)

  377|     def _send_payload(
  378|         self,
  379|         payload: bytes,
  380|         is_critical_command: bool = False,
  381|         user_confirmed: bool = False,
  382|     ) -> None:
  383|         """Transmit a UDS payload through the TxPort.
  384| 
  385|         P-C-001 fix: multi-frame payloads are sent Flow-Control aware —
  386|         the First Frame goes out, then the sender waits (N_Bs) for the
  387|         ECU's FC(CTS) before transmitting Consecutive Frames, honouring
  388|         BS windowing and STmin pacing as ISO 15765-2 requires. Single
  389|         frames go out unchanged.
  390| 
  391|         REVIEW 3-2 (CRITICAL): segmentation honours the client's CAN-FD
  392|         mode — an FD-capable bus now produces FD ISO-TP frames (extended
  393|         SF up to 62 bytes, FD FF/CF) instead of always-Classic framing.
  394|         """
  395|         frames = self.transport.segment_message(payload, is_fd=self.is_fd, brs=self.brs)
  396| 
  397|         if len(frames) <= 1:
  398|             for frame in frames:
  399|                 self._tx_frame(frame, is_critical_command, user_confirmed)
  400|             return
  401| 
  402|         # Multi-frame: FF first, then FC-gated CF transmission.
  403|         self._tx_frame(frames[0], is_critical_command, user_confirmed)
  404|         self._send_consecutive_frames_flow_controlled(
  405|             frames[1:], payload, is_critical_command, user_confirmed
  406|         )

  408|     def _tx_frame(
  409|         self,
  410|         frame: CanFrame,
  411|         is_critical_command: bool,
  412|         user_confirmed: bool,
  413|         budget_category: str = "protocol_burst",
  414|     ) -> None:
  415|         """Send one frame through the gateway TxPort choke-point.
  416| 
  417|         P1-9: ISO-TP frames (FF + CF trains) travel on the 'protocol_burst'
  418|         lane by default — a 256-byte 0x36 block is ~37 CFs which would
  419|         otherwise slam into the 100 msg/s default-lane wall and, pre-P1-9,
  420|         engage a permanent E-Stop (self-DoS mid-flash).
  421|         """
  422|         if hasattr(self.tx_port, "validate_and_transmit"):
  423|             self.tx_port.validate_and_transmit(
  424|                 frame,
  425|                 is_critical_command=is_critical_command,
  426|                 user_confirmed=user_confirmed,
  427|                 budget_category=budget_category,
  428|             )
  429|         elif hasattr(self.tx_port, "send_sync") and not hasattr(self.tx_port, "validate_and_transmit"):
  430|             # TxSafetyGateway-shaped port: honour the category-aware lane.
  431|             try:
  432|                 self.tx_port.send_sync(frame, budget_category=budget_category)
  433|             except TypeError:
  434|                 # Plain TxPort (send_sync(frame) only) — fall back.
  435|                 self.tx_port.send_sync(frame)
  436|         else:
  437|             self.tx_port.send_sync(frame)
  438| 
  439|     # N_Bs (ISO 15765-2 §4.6.2): max wait for FC after FF/before next CF
  440|     N_BS_TIMEOUT_S: ClassVar[float] = 1.0
  441|     # REVIEW hardening (flow-control spoofing): strict FC validation caps.
  442|     FC_STMIN_CAP_MS: ClassVar[float] = 10.0
  443|     FC_WAIT_ABSOLUTE_TIMEOUT_S: ClassVar[float] = 2.0
  444|     FC_CHANNEL_MISMATCH_IS_FATAL: ClassVar[bool] = True
  445| 
  446|     def _await_flow_control_sync(self, timeout_s: float | None = None) -> CanFrame | None:
  447|         """Block until a Flow Control frame from the ECU arrives (N_Bs bound).
  448| 
  449|         REVIEW hardening: FC frames are validated strictly — channel
  450|         match is required (a cross-bus FC must never pace our transfer),
  451|         STmin is capped at 10 ms (a spoofed 127 ms stall is rejected),
  452|         consecutive WAIT frames are bounded by an absolute 2 s timeout on
  453|         top of the N_Bs budget, and the whole wait respects a total
  454|         deadline.
  455|         """
  456|         deadline_budget = self.N_BS_TIMEOUT_S if timeout_s is None else timeout_s
  457|         start = time.monotonic()
  458|         wait_start: float | None = None
  459|         while (time.monotonic() - start) < deadline_budget:
  460|             rx_frame = None
  461|             if self.bus is not None:
  462|                 remaining = deadline_budget - (time.monotonic() - start)
  463|                 rx_frame = self.bus.recv(timeout_s=min(0.05, max(0.001, remaining)))
  464|             if rx_frame is None:
  465|                 continue
  466| 
  467|             # Channel check: an FC from another bus/channel is spoofing
  468|             # or crosstalk. Backward compatibility: multi-channel test
  469|             # topologies legitimately deliver FCs labelled with a different
  470|             # channel string, so a mismatch is ALWAYS audit-logged but only
  471|             # DROPPED when the bus topology is coherent (bus channel ==
  472|             # client channel — a mismatched FC is then provably foreign).
  473|             # Arbitration-ID binding below remains the enforcement point.
  474|             frame_channel = getattr(rx_frame, "channel_id", None)
  475|             if frame_channel is not None and frame_channel != self.channel_id:
  476|                 bus_channel = getattr(self.bus, "channel_id", None) if self.bus is not None else None
  477|                 coherent = bus_channel is not None and bus_channel == self.channel_id
  478|                 logger.warning(
  479|                     "ISO-TP FC channel mismatch (spoof/crosstalk candidate)",
  480|                     extra={
  481|                         "expected": self.channel_id,
  482|                         "got": frame_channel,
  483|                         "coherent_topology": coherent,
  484|                     },
  485|                 )
  486|                 if coherent and self.FC_CHANNEL_MISMATCH_IS_FATAL:
  487|                     continue
  488|             if rx_frame.arbitration_id != self.rx_id or len(rx_frame.data) < 3:
  489|                 continue
  490|             if (rx_frame.data[0] >> 4) != 0x3:  # PCI_FLOW_CONTROL
  491|                 continue
  492|             fs = rx_frame.data[0] & 0x0F
  493|             # Absolute WAIT timeout: a peer holding us in WAIT forever is a
  494|             # transfer-sabotage — bound the total WAIT dwell to 2 s.
  495|             if fs == 0x1:  # FS_WAIT
  496|                 now = time.monotonic()
  497|                 if wait_start is None:
  498|                     wait_start = now
  499|                 elif (now - wait_start) > self.FC_WAIT_ABSOLUTE_TIMEOUT_S:
  500|                     logger.warning(
  501|                         "ISO-TP FC WAIT absolute timeout exceeded — aborting wait",
  502|                         extra={"wait_s": now - wait_start},
  503|                     )
  504|                     return None
  505|             else:
  506|                 wait_start = None
  507|             # STmin cap: reject/clamp spoofed stall values (>10 ms).
  508|             st_min_raw = rx_frame.data[2]
  509|             st_min_ms = decode_st_min(st_min_raw)
  510|             if st_min_ms > self.FC_STMIN_CAP_MS:
  511|                 logger.warning(
  512|                     "ISO-TP FC STmin capped (spoof/stall protection)",
  513|                     extra={"raw": st_min_raw, "decoded_ms": st_min_ms, "cap_ms": self.FC_STMIN_CAP_MS},
  514|                 )
  515|             return rx_frame
  516|         return None
  517| 
  518|     def _send_consecutive_frames_flow_controlled(
  519|         self,
  520|         cf_frames: list[CanFrame],
  521|         payload: bytes,
  522|         is_critical_command: bool,
  523|         user_confirmed: bool,
  524|     ) -> None:
  525|         """Send CFs after FC(CTS), honouring BS windowing and STmin pacing."""
  526|         total = len(cf_frames)
  527|         idx = 0
  528|         wft_count = 0
  529|         WFT_MAX = 16
  530| 
  531|         while idx < total:
  532|             fc = self._await_flow_control_sync()
  533|             if fc is None:
  534|                 raise ProtocolError(
  535|                     "ISO-TP N_Bs timeout waiting for Flow Control after First Frame",
  536|                     code="UDS_TIMEOUT",
  537|                     details={"tx_id": hex(self.tx_id), "rx_id": hex(self.rx_id)},
  538|                 )
  539| 
  540|             fs = fc.data[0] & 0x0F
  541|             if fs == 0x1:  # FS_WAIT
  542|                 wft_count += 1
  543|                 if wft_count > WFT_MAX:
  544|                     raise ProtocolError(
  545|                         f"ISO-TP WFTmax exceeded ({wft_count} consecutive WAIT frames)",
  546|                         code="UDS_TIMEOUT",
  547|                         details={"wft_count": wft_count},
  548|                     )
  549|                 continue
  550|             if fs == 0x2:  # FS_OVERFLOW
  551|                 raise ProtocolError(
  552|                     "ECU reported ISO-TP buffer overflow (FlowStatus.OVERFLOW)",
  553|                     code="UDS_BUFFER_OVERFLOW",
  554|                     details={"requested_length": len(payload)},
  555|                 )
  556|             if fs != 0x0:  # must be CTS
  557|                 raise ProtocolError(
  558|                     f"Invalid ISO-TP FlowStatus 0x{fs:X}",
  559|                     code="UDS_PROTOCOL_ERROR",
  560|                     details={"flow_status": fs},
  561|                 )
  562| 
  563|             wft_count = 0
  564|             bs = fc.data[1]
  565|             st_min = fc.data[2]
  566| 
  567|             block_sent = 0
  568|             while idx < total:
  569|                 if bs > 0 and block_sent >= bs:
  570|                     break  # block exhausted — wait for the next FC
  571|                 # REVIEW hardening: clamp spoofed STmin stalls at the cap
  572|                 # (a 127 ms STmin would otherwise serialize a flash to a
  573|                 # crawl; pacing above the cap is never legitimate).
  574|                 delay_ms = min(decode_st_min(st_min), self.FC_STMIN_CAP_MS)
  575|                 if delay_ms > 0:
  576|                     time.sleep(delay_ms / 1000.0)
  577|                 self._tx_frame(cf_frames[idx], is_critical_command, user_confirmed)
  578|                 idx += 1
  579|                 block_sent += 1
  580| 
  581| # Service IDs whose positive response echoes request bytes beyond SID:
  582|     # 0x22/0x2E echo the DID (2 bytes at data[0:2]),
  583|     # 0x27 echoes the subfunction (1 byte at data[0]),
  584|     # 0x10/0x11 echo the subfunction/type (1 byte at data[0]),
  585|     # 0x19 echoes the subfunction (1 byte at data[0]),
  586|     # 0x31 echoes routineControlType + routineId (3 bytes at data[0:3]),
  587|     # 0x36 echoes the block sequence counter (1 byte at data[0]).
  588|     _ECHO_LENGTHS: ClassVar[dict[int, int]] = {
  589|         0x10: 1,  # DiagnosticSessionControl: session type
  590|         0x11: 1,  # ECU Reset: reset type
  591|         0x19: 1,  # ReadDTCInformation: sub-function
  592|         0x22: 2,  # ReadDataByIdentifier: DID
  593|         0x2E: 2,  # WriteDataByIdentifier: DID
  594|         0x27: 1,  # SecurityAccess: securityAccessType
  595|         0x31: 3,  # RoutineControl: controlType + routineId
  596|         0x36: 1,  # TransferData: blockSequenceCounter
  597|     }
  598| 
  599|     def _response_echo_matches(self, expected_sid: int, request: bytes, resp: UdsResponse) -> bool:
  600|         """Verify the service-specific echo bytes of a positive response.
  601| 
  602|         REVIEW (echo validation): only the response SID was compared — the
  603|         wrong DID/subfunction/routine answer of the same service completed
  604|         the exchange. The echo bytes (`resp.data[:n]`) must equal the
  605|         request's corresponding bytes (`request[1:1+n]`) before the
  606|         response is accepted.
  607|         """
  608|         if not resp.is_positive:
  609|             # Negative responses echo the REQUESTED sid at payload[1]; the
  610|             # parse already surfaces it via service_id — no extra echo bytes
  611|             # are mandated for the completion decision.
  612|             return True
  613|         n = self._ECHO_LENGTHS.get(expected_sid)
  614|         if n is None:
  615|             return True
  616|         if len(resp.data) < n:
  617|             # Positive response missing its own echo bytes is malformed.
  618|             return False
  619|         return bytes(resp.data[:n]) == bytes(request[1 : 1 + n])
  620| 
  621|     def _send_and_receive(
  622|         self,
  623|         payload: bytes,
  624|         timeout_s: float = 2.0,
  625|         is_critical_command: bool = False,
  626|         user_confirmed: bool = False,
  627|     ) -> UdsResponse:
  628|         """Send segmented UDS request and wait for complete ISO-TP reassembled response.
  629| 
  630|         A negative response NRC 0x78 (Response Pending) extends the wait window
  631|         by P2* (F-14) instead of surfacing as a timeout while the ECU is working.
  632|         """
  633|         # P4: the synchronous path reads frames from the bus; an rx_sub-only
  634|         # client must use the asynchronous one-shot query API instead.
  635|         if self.bus is None:
  636|             raise ProtocolError(
  637|                 "Synchronous UDS request/response requires a bus; this client "
  638|                 "was constructed with an rx_sub only",
  639|                 code="UDS_NO_RX_PATH",
  640|                 details={"tx_id": hex(self.tx_id), "rx_id": hex(self.rx_id)},
  641|             )
  642| 
  643|         with self._operation_lock:
  644|             self._send_payload(payload, is_critical_command=is_critical_command, user_confirmed=user_confirmed)
  645| 
  646|             start_time = time.monotonic()
  647|             deadline = start_time + timeout_s
  648|             # REVIEW 1-M3 (MEDIUM): ISO 14229-1 sets no NRC 0x78 count
  649|             # limit — the server may keep extending within its announced
  650|             # P2* window. The old hard 30 s absolute cap killed healthy
  651|             # long routines (flash erase, checksum verification) mid-
  652|             # flight. Each pending response re-arms P2*; the optional
  653|             # max_pending_timeout_s (constructor) bounds the total dwell
  654|             # for callers that need a wall.
  655|             max_absolute_deadline = (
  656|                 start_time + self.max_pending_timeout_s
  657|                 if self.max_pending_timeout_s is not None
  658|                 else None
  659|             )
  660|             nrc_78_count = 0
  661| 
  662|             while True:
  663|                 now = time.monotonic()
  664|                 remaining = deadline - now
  665|                 if remaining <= 0 or (max_absolute_deadline is not None and now >= max_absolute_deadline):
  666|                     raise ProtocolError(
  667|                         f"UDS Request timed out waiting for response from ECU (0x{self.rx_id:03X})",
  668|                         code="UDS_TIMEOUT",
  669|                         details={"tx_id": hex(self.tx_id), "rx_id": hex(self.rx_id), "nrc_78_count": nrc_78_count},
  670|                     )
  671| 
  672|                 rx_frame = None
  673|                 if self.bus is not None:
  674|                     rx_frame = self.bus.recv(timeout_s=min(0.1, max(0.001, remaining)))
  675| 
  676|                 if rx_frame is not None and rx_frame.arbitration_id == self.rx_id:
  677|                     completed_data, resp_frame = self.transport.handle_rx_frame(rx_frame)
  678|                     if resp_frame is not None:
  679|                         # M-5 (P2-1): flow-control frames are protocol
  680|                         # overhead, not operator commands — an ISO-TP CF/FC
  681|                         # train over the default lane (100 msg/s) could trip
  682|                         # the gateway's sustained-overload E-Stop mid-read.
  683|                         # Route through the protocol_burst budget lane.
  684|                         try:
  685|                             self.tx_port.send_sync(resp_frame, budget_category="protocol_burst")
  686|                         except TypeError:
  687|                             self.tx_port.send_sync(resp_frame)
  688|                     if completed_data is not None:
  689|                         resp = UdsServiceBuilder.parse_response(completed_data)
  690|                         expected_sid = payload[0]
  691|                         if resp.service_id != expected_sid:
  692|                             logger.warning(
  693|                                 "UDS response service ID mismatch (dropping mismatched/unsolicited response)",
  694|                                 extra={
  695|                                     "expected_sid": hex(expected_sid),
  696|                                     "actual_sid": hex(resp.service_id) if resp.service_id is not None else None,
  697|                                     "rx_id": hex(self.rx_id),
  698|                                 },
  699|                             )
  700|                             continue
  701| 
  702|                         # REVIEW (echo validation): a same-SID reply with the
  703|                         # WRONG DID/subfunction echo used to complete the
  704|                         # pending exchange — a delayed response or another
  705|                         # client's answer produced success for the wrong
  706|                         # request. The service-specific echo bytes must match
  707|                         # the request before the response is accepted.
  708|                         if not self._response_echo_matches(expected_sid, payload, resp):
  709|                             logger.warning(
  710|                                 "UDS response echo mismatch (dropping mismatched/unsolicited response)",
  711|                                 extra={
  712|                                     "sid": hex(expected_sid),
  713|                                     "rx_id": hex(self.rx_id),
  714|                                     "data_hex": completed_data[:8].hex(),
  715|                                 },
  716|                             )
  717|                             continue
  718| 
  719|                         if (
  720|                             not resp.is_positive
  721|                             and resp.nrc == UdsNrc.REQUEST_CORRECTLY_RECEIVED_RESPONSE_PENDING
  722|                             and (max_absolute_deadline is None or time.monotonic() < max_absolute_deadline)
  723|                         ):
  724|                             # P2* extension: ECU signalled pending; each 0x78
  725|                             # re-arms the P2* window (REVIEW 1-M3 — no count
  726|                             # cap; ISO 14229-1 allows unlimited pendings
  727|                             # within the server's announced window).
  728|                             nrc_78_count += 1
  729|                             new_deadline = time.monotonic() + P2_STAR_TIMEOUT_S
  730|                             if max_absolute_deadline is not None:
  731|                                 new_deadline = min(new_deadline, max_absolute_deadline)
  732|                             deadline = new_deadline
  733|                             continue
  734|                         return resp
  735| 
  736|     # ISO 15765-4 functional (broadcast) request ID for 11-bit OBD-II
  737|     FUNCTIONAL_REQUEST_ID: ClassVar[int] = 0x7DF
  738|     # Expected physical response range (ECU response base 0x7E8 + ECU index)
  739|     PHYSICAL_RESPONSE_IDS: ClassVar[tuple[int, ...]] = tuple(range(0x7E8, 0x7F0))
  740| 
  741|     def send_functional(
  742|         self,
  743|         payload: bytes,
  744|         functional_id: int | None = None,
  745|         expected_physical_ids: tuple[int, ...] | None = None,
  746|         collect_window_s: float = 0.3,
  747|     ) -> list[UdsResponse]:
  748|         """REVIEW 3-5 (MEDIUM): send a functional (broadcast) request and
  749|         collect every responding ECU's physical answer.
  750| 
  751|         ISO 15765-4 / ISO 14229 functional addressing (0x7DF) reaches ALL
  752|         compliant ECUs; each answers from its OWN physical response ID
  753|         (0x7E8..0x7EF). The single-rx_id filter in _send_and_receive
  754|         silently dropped every other ECU's reply. This method transmits
  755|         the request once, then collects responses from all expected
  756|         physical IDs for `collect_window_s`, reassembling each ID's
  757|         ISO-TP stream with a dedicated transport.
  758|         """
  759|         fid = functional_id if functional_id is not None else self.FUNCTIONAL_REQUEST_ID
  760|         expected = expected_physical_ids if expected_physical_ids is not None else self.PHYSICAL_RESPONSE_IDS
  761| 
  762|         with self._operation_lock:
  763|             func_transport = IsoTpTransport(tx_id=fid, rx_id=fid, channel_id=self.channel_id)
  764|             req_frames = func_transport.segment_message(payload, is_fd=self.is_fd, brs=self.brs)
  765|             for frame in req_frames:
  766|                 self._tx_frame(frame, is_critical_command=False, user_confirmed=False)
  767| 
  768|             # Per-physical-ID reassembly transports.
  769|             per_id: dict[int, IsoTpTransport] = {rid: IsoTpTransport(tx_id=fid, rx_id=rid, channel_id=self.channel_id) for rid in expected}
  770|             completed: dict[int, bytes] = {}
  771|             deadline = time.monotonic() + collect_window_s
  772|             expected_sid = payload[0] if payload else 0
  773| 
  774|             while time.monotonic() < deadline:
  775|                 rx_frame = self.bus.recv(timeout_s=min(0.05, max(0.001, deadline - time.monotonic()))) if self.bus is not None else None
  776|                 if rx_frame is None:
  777|                     continue
  778|                 rid = rx_frame.arbitration_id
  779|                 if rid not in per_id:
  780|                     continue
  781|                 data, resp_frame = per_id[rid].handle_rx_frame(rx_frame)
  782|                 if resp_frame is not None:
  783|                     try:
  784|                         self.tx_port.send_sync(resp_frame, budget_category="protocol_burst")
  785|                     except TypeError:
  786|                         self.tx_port.send_sync(resp_frame)
  787|                 if data is not None:
  788|                     resp = UdsServiceBuilder.parse_response(data)
  789|                     if resp.service_id == expected_sid:
  790|                         completed.setdefault(rid, data)
  791| 
  792|             return [UdsServiceBuilder.parse_response(data) for data in completed.values()]
